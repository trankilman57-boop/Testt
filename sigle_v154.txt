Private Function ConstruireDictionnaireComplet(ByRef tblDebuts() As Long, ByRef tblFins() As Long, ByRef tblColSigle() As Long, ByRef nbTables As Long) As Object
    Dim dico As Object
    Set dico = CreateObject("Scripting.Dictionary")
    dico.CompareMode = vbTextCompare

    ChargerDicoDepuisTableaux dico, tblDebuts, tblFins, tblColSigle, nbTables

    ' [v93] Le v92 gardait une définition manuelle "parce
    ' qu'utilisée dans le texte" — mais c'est justement TOUJOURS
    ' vrai pour un sigle déjà lié, rendant cette vérification
    ' inutile en pratique (aucun effet réel constaté). Simplifié :
    ' le TABLEAU est désormais entièrement prioritaire/autoritaire.
    ' Toute définition manuelle absente du tableau actuel est
    ' purgée systématiquement, sans exception — une ligne supprimée
    ' du tableau n'est donc plus jamais reconnue au traitement
    ' suivant, même si le sigle apparaît encore dans le texte.
    Dim dicoDefM As Object
    Set dicoDefM = ChargerDefinitionsManuelles()
    Dim cleD As Variant
    For Each cleD In dicoDefM.Keys
        If dico.Exists(cleD) Then
            ' Déjà présent via le tableau : le tableau prime, rien à faire.
        Else
            SupprimerDefinitionManuelle CStr(cleD)
        End If
    Next cleD

    Set ConstruireDictionnaireComplet = dico
End Function

' [v87] Construit le dictionnaire à partir d'un tableau désigné
' MANUELLEMENT par l'utilisateur via SÉLECTION (plutôt que par
' saisie du texte exact de l'en-tête, sujette aux fautes de
' frappe). L'utilisateur sélectionne d'abord la cellule d'en-tête
' de la colonne sigle, puis celle de la colonne signification ;
' toutes les lignes de données du même tableau sont ensuite lues
' directement via leur ColumnIndex réel.
Private Function ChargerDicoDepuisSelection(ByRef tblDebuts() As Long, ByRef tblFins() As Long, ByRef tblColSigle() As Long, ByRef nbTables As Long) As Object

    Dim dico As Object
    Set dico = CreateObject("Scripting.Dictionary")
    dico.CompareMode = vbTextCompare
    nbTables = 0

    ' [v88] Plus aucun MsgBox intermédiaire (bloquait la sélection
    ' dans le document, MsgBox étant modal). La sélection des 2
    ' cellules d'en-tête (sigle à gauche, signification à droite)
    ' doit être faite AVANT le lancement de la macro — on lit ici
    ' directement Selection, déjà positionnée par l'utilisateur.
    Dim dansTableauSel As Boolean
    dansTableauSel = False
    On Error Resume Next
    dansTableauSel = Selection.Information(wdWithInTable)
    On Error GoTo 0

    If Not dansTableauSel Or Selection.Cells.Count < 2 Then
        Set ChargerDicoDepuisSelection = dico
        Exit Function
    End If

    ' --- Détermine la cellule la plus à GAUCHE (sigle) et la plus
    ' à DROITE (signification) de la sélection ---
    Dim tbl As Table
    Dim celSel As Cell
    Dim colSigleSel As Long
    Dim colSignifSel As Long
    colSigleSel = 0
    colSignifSel = 0

    On Error Resume Next
    Set tbl = Selection.Tables(1)
    For Each celSel In Selection.Cells
        If colSigleSel = 0 Or celSel.ColumnIndex < colSigleSel Then colSigleSel = celSel.ColumnIndex
        If celSel.ColumnIndex > colSignifSel Then colSignifSel = celSel.ColumnIndex
    Next celSel
    On Error GoTo 0

    If tbl Is Nothing Or colSigleSel = 0 Or colSignifSel = 0 Or colSigleSel = colSignifSel Then
        Set ChargerDicoDepuisSelection = dico
        Exit Function
    End If

    ' --- Confirmation avant de poursuivre ---
    Dim apercuSigle As String
    Dim apercuSignif As String
    On Error Resume Next
    apercuSigle = NettoyerTexteCellule(tbl.Cell(1, colSigleSel).Range.Text)
    apercuSignif = NettoyerTexteCellule(tbl.Cell(1, colSignifSel).Range.Text)
    On Error GoTo 0

    Dim repConfirm As VbMsgBoxResult
    repConfirm = MsgBox( _
        "Colonne SIGLE détectée : """ & apercuSigle & """" & vbCrLf & _
        "Colonne SIGNIFICATION détectée : """ & apercuSignif & """" & vbCrLf & vbCrLf & _
        "Confirmer et charger ce tableau ?" & vbCrLf & _
        "(Non = recommencer : resélectionnez les 2 cellules d'en-tête puis relancez la macro)", _
        vbYesNo + vbQuestion, "Confirmation du tableau de sigles")

    If repConfirm = vbNo Then
        Set ChargerDicoDepuisSelection = dico
        Exit Function
    End If

    ' [v95] Mémorise les intitulés détectés par sélection, comme
    ' pour la saisie au clavier (v76) — sans ça,
    ' TrouverTableauSiglesPourAjout (utilisée par l'ajout ponctuel
    ' via AjouterDefinitionDepuisSelection) ne pouvait jamais
    ' retrouver ce tableau, puisqu'il n'avait été désigné que par
    ' sélection, sans laisser aucune trace persistante.
    If Len(apercuSigle) > 0 And Len(apercuSignif) > 0 Then
        On Error Resume Next
        ActiveDocument.Variables(NOM_VARIABLE_INTITULE_SIGLE).Delete
        ActiveDocument.Variables(NOM_VARIABLE_INTITULE_SIGNIF).Delete
        On Error GoTo 0
        ActiveDocument.Variables.Add NOM_VARIABLE_INTITULE_SIGLE, apercuSigle
        ActiveDocument.Variables.Add NOM_VARIABLE_INTITULE_SIGNIF, apercuSignif
    End If

    ' --- Lecture de toutes les lignes de données du tableau désigné ---
    Dim cel As Cell
    Dim rIdxSel As Long
    Dim sigle As String
    Dim signif As String

    For Each cel In tbl.Range.Cells
        rIdxSel = cel.RowIndex
        If rIdxSel >= 2 Then
            If cel.ColumnIndex = colSigleSel Then
                sigle = NettoyerTexteCellule(cel.Range.Text)
            ElseIf cel.ColumnIndex = colSignifSel Then
                signif = NettoyerTexteCellule(cel.Range.Text)
                If Len(signif) > 80 Then signif = Left(signif, 80)
                If Len(sigle) >= 2 And Len(sigle) <= 15 And Len(signif) >= 2 Then
                    If Not dico.Exists(sigle) Then dico.Add sigle, signif
                End If
            End If
        End If
    Next cel

    ' --- Enregistre ce tableau dans les tableaux de bornes (exclusion colonne sigle) ---
    ReDim tblDebuts(0 To 0)
    ReDim tblFins(0 To 0)
    ReDim tblColSigle(0 To 0)
    tblDebuts(0) = tbl.Range.Start
    tblFins(0) = tbl.Range.End
    tblColSigle(0) = colSigleSel
    nbTables = 1

    Set ChargerDicoDepuisSelection = dico

End Function

' [v81] FACTORISATION : détermine si un mot situé dans un
' tableau se trouve précisément dans la colonne "sigle" (donc à
' exclure). Remplace un bloc dupliqué dans TraiterDocumentUnePasse
' et SurlignerRougeSeul.
Private Function EstDansColonneSigleExclue(posMotStart As Long, wrd As Range, tblDebuts() As Long, tblFins() As Long, tblColSigle() As Long, nbTables As Long) As Boolean
    Dim tIdx As Long
    EstDansColonneSigleExclue = False
    For tIdx = 0 To nbTables - 1
        If posMotStart >= tblDebuts(tIdx) And posMotStart <= tblFins(tIdx) Then
            Dim colActuelle As Long
            On Error Resume Next
            colActuelle = wrd.Cells(1).ColumnIndex
            On Error GoTo 0
            EstDansColonneSigleExclue = (colActuelle = tblColSigle(tIdx))
            Exit Function
        End If
    Next tIdx
End Function

' [v89] Détecte les sigles du dictionnaire qui n'apparaissent
' JAMAIS dans le corps du document (hors des tableaux de sigles
' eux-mêmes) — candidats à la suppression du tableau de
' référence. Retourne un Dictionary des sigles inutilisés.
' [v113] Vrai si la plage trouvée est un MOT isolé : caractère avant
' non alphanumérique (l'apostrophe, le tiret, l'espace, la ponctuation
' sont acceptés) et caractère après non alphanumérique, sauf un "s"
' de pluriel suivi d'un non-alphanumérique (ONGs). Évite de
' considérer "ONG" utilisé parce que "ONGLET" ou "CONGRES" existe.
Private Function EstCaractereMot(c As String) As Boolean
    If Len(c) = 0 Then
        EstCaractereMot = False
    Else
        EstCaractereMot = ((c >= "A" And c <= "Z") Or (c >= "a" And c <= "z") Or (c >= "0" And c <= "9") Or AscW(c) >= 192)
    End If
End Function

Private Function OccurrenceIsolee(rng As Range) As Boolean
    Dim cAv As String, cAp As String, cAp2 As String
    OccurrenceIsolee = True
    On Error Resume Next
    If rng.Start > 0 Then cAv = ActiveDocument.Range(rng.Start - 1, rng.Start).Text
    cAp = ActiveDocument.Range(rng.End, rng.End + 1).Text
    cAp2 = ActiveDocument.Range(rng.End + 1, rng.End + 2).Text
    On Error GoTo 0
    If EstCaractereMot(cAv) Then
        OccurrenceIsolee = False
        Exit Function
    End If
    If EstCaractereMot(cAp) Then
        If cAp = "s" And Not EstCaractereMot(cAp2) Then
            OccurrenceIsolee = True
        Else
            OccurrenceIsolee = False
        End If
    End If
End Function

' [v114/v115] Vrai si la plage est dans la cellule DÉFINITION d'une
' ligne d'un tableau de sigles dont le sigle est DIFFÉRENT du sigle
' cherché : un sigle cité dans la définition d'un autre sigle est
' considéré comme utilisé. Exclusions POSITIVES seulement (ligne de
' titre, colonne sigle, ligne du sigle lui-même) ; en cas de doute
' ou d'erreur (plage dans un champ/lien, cellules fusionnées) on
' répond Vrai = on CONSERVE le sigle (jamais de suppression à tort).
Private Function OccurrenceDansDefinitionAutreSigle(rng As Range, sigle As String, colSigle As Long) As Boolean
    Dim ce As Cell
    Dim ceLigne As Cell
    Dim txtSigleLigne As String

    OccurrenceDansDefinitionAutreSigle = True   ' prudence : on garde

    Set ce = Nothing
    On Error Resume Next
    Set ce = rng.Cells(1)
    If ce Is Nothing Then Set ce = ActiveDocument.Range(rng.Start, rng.Start).Cells(1)
    On Error GoTo 0
    If ce Is Nothing Then Exit Function

    On Error Resume Next
    If ce.RowIndex <= 1 Then
        OccurrenceDansDefinitionAutreSigle = False
        Exit Function
    End If
    If ce.ColumnIndex = colSigle Then
        OccurrenceDansDefinitionAutreSigle = False
        Exit Function
    End If
    txtSigleLigne = ""
    For Each ceLigne In ce.Row.Cells
        If ceLigne.ColumnIndex = colSigle Then
            txtSigleLigne = Trim(NettoyerTexteCellule(ceLigne.Range.Text))
            Exit For
        End If
    Next ceLigne
    If Len(txtSigleLigne) > 0 Then
        If StrComp(txtSigleLigne, sigle, vbTextCompare) = 0 Then OccurrenceDansDefinitionAutreSigle = False
    End If
    On Error GoTo 0
End Function

Private Function DetecterSiglesInutilises(dico As Object, tblDebuts() As Long, tblFins() As Long, tblColSigle() As Long, nbTables As Long) As Object

    ' [v103] Recherche par Find (au lieu de mot Word par mot Word) :
    ' corrige les faux "inutilisés" (sigles à tiret, pluriels, ponctuation).
    Dim inutilises As Object
    Set inutilises = CreateObject("Scripting.Dictionary")
    inutilises.CompareMode = vbTextCompare

    Dim cleU As Variant
    Dim rngU As Range
    Dim trouveU As Boolean
    Dim dansTableU As Boolean
    Dim tIdxU As Long
    Dim nbU As Long

    For Each cleU In dico.Keys
        nbU = nbU + 1
        If nbU Mod 10 = 0 Then
            Application.StatusBar = "Recherche des sigles inutilisés : " & nbU & "/" & dico.Count
            DoEvents
        End If

        trouveU = False
        Set rngU = ActiveDocument.Content
        With rngU.Find
            .ClearFormatting
            .Text = CStr(cleU)
            .MatchCase = True
            .MatchWholeWord = False
            .MatchWildcards = False
            .Forward = True
            .Wrap = wdFindStop
        End With

        Do While rngU.Find.Execute
            dansTableU = False
            For tIdxU = 0 To nbTables - 1
                If rngU.Start >= tblDebuts(tIdxU) And rngU.Start <= tblFins(tIdxU) Then
                    dansTableU = True
                    Exit For
                End If
            Next tIdxU
            ' [v114] Dans un tableau de sigles : ne compte que si c'est dans
            ' la définition d'un AUTRE sigle (pas colonne sigle, pas titre,
            ' pas sa propre ligne).
            If dansTableU Then
                If Not OccurrenceDansDefinitionAutreSigle(rngU, CStr(cleU), tblColSigle(tIdxU)) Then GoTo OccurrenceSuivante
            End If
            ' [v113] occurrence isolée seulement (pas ONG dans ONGLET)
            If OccurrenceIsolee(rngU) Then
                trouveU = True
                Exit Do
            End If
OccurrenceSuivante:
            rngU.Collapse wdCollapseEnd
        Loop

        If Not trouveU Then inutilises.Add cleU, dico(cleU)
    Next cleU

    Set DetecterSiglesInutilises = inutilises

End Function

    t = Replace(t, Chr(13), "")
    t = Replace(t, Chr(7), "")

    t = Replace(t, Chr(9), "")
    Do While InStr(t, "  ") > 0
        t = Replace(t, "  ", " ")
    Loop
    NettoyerTexteCellule = Trim(t)
End Function

' Vérifie qu'une chaîne ressemble à un sigle :


Private Function EstUnSigle(s As String) As Boolean
    Dim i As Long
    Dim c As String
    EstUnSigle = False
    If Len(s) < 2 Or Len(s) > 10 Then Exit Function
    For i = 1 To Len(s)
        c = Mid(s, i, 1)
        If Not ((c >= "A" And c <= "Z") Or (c >= "0" And c <= "9")) Then Exit Function
    Next i
    Dim aUneLettre As Boolean
    aUneLettre = False
    For i = 1 To Len(s)
        c = Mid(s, i, 1)
        If c >= "A" And c <= "Z" Then
            aUneLettre = True
            Exit For
        End If
    Next i
    EstUnSigle = aUneLettre
End Function
' Récupère le dernier "mot" avant une parenthèse ouvrante


Private Function ExtraireDernierMot(texteAvant As String) As String
    Dim t As String
    Dim i As Long
    Dim c As String
    Dim debut As Long
    t = texteAvant
    Do While Len(t) > 0 And Right(t, 1) = " "
        t = Left(t, Len(t) - 1)
    Loop
    debut = Len(t)
    For i = Len(t) To 1 Step -1
        c = Mid(t, i, 1)
        If c = " " Or c = vbCr Or c = vbLf Or c = vbTab Then Exit For
        debut = i
    Next i
    ExtraireDernierMot = Mid(t, debut)
End Function
' Remonte dans le texte avant la parenthèse pour extraire


Private Function ExtraireGroupeNominal(texteAvant As String) As String
    Dim texte As String
    Dim posDebut As Long
    Dim posPoint As Long
    Dim posVirgPoint As Long
    Dim posRetourCr As Long
    Dim posRetourLf As Long
    Dim limite As Long
    Dim candidat As String
    texte = texteAvant
    posDebut = Len(texte) - 120
    If posDebut < 1 Then posDebut = 1
    texte = Mid(texte, posDebut)
    posPoint = InStrRev(texte, ". ")
    posVirgPoint = InStrRev(texte, "; ")
    posRetourCr = InStrRev(texte, vbCr)
    posRetourLf = InStrRev(texte, vbLf)
    limite = posPoint
    If posVirgPoint > limite Then limite = posVirgPoint
    If posRetourCr > limite Then limite = posRetourCr
    If posRetourLf > limite Then limite = posRetourLf
    candidat = Trim(Mid(texte, limite + 1))
    candidat = Replace(candidat, vbCr, "")
    candidat = Replace(candidat, vbLf, "")
    If Len(candidat) > 80 Then candidat = Left(candidat, 80)
    If Len(candidat) >= 3 And Len(candidat) <= 80 Then
        If UCase(Left(candidat, 1)) = Left(candidat, 1) Then
            ExtraireGroupeNominal = candidat
            Exit Function
        End If
    End If
    ExtraireGroupeNominal = ""
End Function
' Vérifie si la plage se trouve dans un titre (Titre/Heading)
' ou dans une table des matières (TOC)


Private Function EstDansTitre(rng As Range) As Boolean
    Dim nomStyle As String
    EstDansTitre = False
    nomStyle = ""

    ' [v25] Tentative 1 : Style directement sur le Range
    ' (plus fiable que Paragraphs(1) sur un Range d'un seul mot)
    On Error Resume Next
    nomStyle = rng.Style.NameLocal
    On Error GoTo 0

    ' [v25] Tentative 2 (repli) : si vide, on essaie via Paragraphs(1)
    If Len(nomStyle) = 0 Then
        On Error Resume Next
        nomStyle = rng.Paragraphs(1).Style.NameLocal
        On Error GoTo 0
    End If

    ' [v25] Tentative 3 (dernier repli) : élargit le Range au
    ' paragraphe entier avant de lire le style
    If Len(nomStyle) = 0 Then
        On Error Resume Next
        Dim rngElargi As Range
        Set rngElargi = rng.Duplicate
        rngElargi.Expand wdParagraph
        nomStyle = rngElargi.Style.NameLocal
        On Error GoTo 0
    End If

    If InStr(1, nomStyle, "Titre", vbTextCompare) > 0 Then
        EstDansTitre = True
        Exit Function
    End If
    If InStr(1, nomStyle, "Heading", vbTextCompare) > 0 Then EstDansTitre = True
    If InStr(1, nomStyle, "TM ", vbTextCompare) > 0 Then EstDansTitre = True
    If InStr(1, nomStyle, "TOC", vbTextCompare) > 0 Then EstDansTitre = True
    If EstDansTitre Then Exit Function

    ' [v26] Détection complémentaire par MISE EN FORME :
    On Error Resume Next
    If rng.Font.Bold = True And rng.Font.Size >= 14 Then
        EstDansTitre = True
    End If
    On Error GoTo 0
End Function

' Charge la liste d'exceptions depuis la variable du document


Private Function ChargerExceptions() As Object
    Dim dicoExcept As Object
    Set dicoExcept = CreateObject("Scripting.Dictionary")
    dicoExcept.CompareMode = vbTextCompare

    On Error Resume Next
    Dim contenuVar As String
    contenuVar = ActiveDocument.Variables(NOM_VARIABLE_EXCEPTIONS).Value
    On Error GoTo 0

    If Len(contenuVar) > 0 Then
        Dim morceaux() As String
        morceaux = Split(contenuVar, SEPARATEUR_EXCEPTIONS)
        Dim k As Long
        For k = LBound(morceaux) To UBound(morceaux)
            Dim m As String
            m = Trim(morceaux(k))
            If Len(m) > 0 Then
                If Not dicoExcept.Exists(m) Then
                    dicoExcept.Add m, True
                End If
            End If
        Next k
    End If

    Set ChargerExceptions = dicoExcept
End Function
' [v73] Charge les définitions ajoutées manuellement (variable de

Private Function ChargerDefinitionsManuelles() As Object
    Dim dicoDef As Object
    Set dicoDef = CreateObject("Scripting.Dictionary")
    dicoDef.CompareMode = vbTextCompare

    On Error Resume Next
    Dim contenuVar As String
    contenuVar = ActiveDocument.Variables(NOM_VARIABLE_DEFINITIONS).Value
    On Error GoTo 0

    If Len(contenuVar) > 0 Then
        Dim entrees() As String
        entrees = Split(contenuVar, SEPARATEUR_DEFINITIONS)
        Dim j As Long
        For j = LBound(entrees) To UBound(entrees)
            Dim entree As String

                    If Len(sigleE) > 0 And Not dicoDef.Exists(sigleE) Then
                        dicoDef.Add sigleE, defE
                    End If

    End If

Private Function TraiterDocumentUnePasse(dico As Object, ByRef rapportDetail As String, ByRef compteurRouge As Long, tblDebuts() As Long, tblFins() As Long, tblColSigle() As Long, nbTables As Long) As Long

    Dim wrd As Range
    Dim motBrut As String
    Dim motCourant As String
    Dim motBase As String
    Dim lenAffiche As Long
    Dim i As Long
    Dim total As Long
    Dim compteur As Long
    Dim nbIgnoreTitre As Long
    Dim nbIgnoreLien As Long
    Dim nbIgnoreAutre As Long

    Dim debuts() As Long
    Dim fins() As Long
    Dim sigleAssocie() As String
    Dim contextes() As String
    Dim nbAColler As Long
    Dim finPropre As Long

    ' [v61] Tableaux pour différer le surlignage rouge (voir plus
    ' bas), même principe que debuts()/fins() pour les liens.
    Dim rougeDebuts() As Long
    Dim rougeFins() As Long
    Dim nbRouge As Long

    Dim comptesTrouve As Object
    Dim comptesLie As Object
    Set comptesTrouve = CreateObject("Scripting.Dictionary")
    comptesTrouve.CompareMode = vbTextCompare
    Set comptesLie = CreateObject("Scripting.Dictionary")
    comptesLie.CompareMode = vbTextCompare

    nbLiensSansEffet = 0
    compteur = 0
    compteurRouge = 0
    i = 0
    nbIgnoreTitre = 0
    nbIgnoreLien = 0
    nbIgnoreAutre = 0
    nbAColler = 0
    nbRouge = 0
    ReDim debuts(0 To 500)
    ReDim fins(0 To 500)
    ReDim sigleAssocie(0 To 500)
    ReDim contextes(0 To 500)
    ReDim rougeDebuts(0 To 500)
    ReDim rougeFins(0 To 500)

    ' [v69] Chargement de la liste d'exceptions (une seule fois,
    Dim dicoExceptions As Object
    Set dicoExceptions = ChargerExceptions()

    total = ActiveDocument.Words.Count

    ' [v41] Regroupe TOUTES les opérations de ce traitement dans
    On Error Resume Next
    Application.UndoRecord.StartCustomRecord "Traitement des sigles"
    On Error GoTo 0

    On Error Resume Next
    Err.Clear
    ActiveDocument.Styles(wdStyleHyperlink).Font.ColorIndex = wdAuto
    Err.Clear
    ActiveDocument.Styles(wdStyleHyperlink).Font.Underline = wdUnderlineNone
    Err.Clear
    ActiveDocument.Styles(wdStyleHyperlink).Font.Bold = False
    Err.Clear
    ActiveDocument.Styles(wdStyleHyperlink).Font.Italic = False
    Err.Clear
    ActiveDocument.Styles(wdStyleHyperlink).Font.Shading.Texture = wdTexture15Percent
    Err.Clear
    ActiveDocument.Styles(wdStyleHyperlink).Font.Shading.ForegroundPatternColor = wdColorAutomatic
    Err.Clear
    ActiveDocument.Styles(wdStyleHyperlink).Font.Shading.BackgroundPatternColor = wdColorAutomatic
    Err.Clear
    On Error GoTo 0

    ' =====================================================
    ' [v55] CORRECTIF DE PERFORMANCE MAJEUR. Depuis v48, la
    ' =====================================================
    Dim wStart() As Long
    Dim wFin() As Long
    Dim wTexte() As String
    Dim totalWords As Long

    ' [v60] Chronométrage par phase, pour localiser précisément où
    ' le temps est passé (Ctrl+G pour voir le détail après coup).
    Dim tMateriali As Single, tClassif As Single, tLiens As Single
    Dim tDebutPhase As Single
    tDebutPhase = Timer

    ' [v81] FACTORISATION : matérialisation via helper commun
    MaterialiserMots wStart, wFin, wTexte, totalWords

    tMateriali = Timer - tDebutPhase
    tDebutPhase = Timer

    Dim idxW As Long
    Dim motSuivant1 As String
    Dim motSuivant2 As String
    Dim motCombine As String
    Dim finCombine As Long
    Dim aFusionne As Boolean

    idxW = 1

    Do While idxW <= totalWords

        i = i + 1
        If i Mod 2000 = 0 Then
            Application.StatusBar = "Analyse du document : " & i & "/" & total & " mots (" & nbAColler & " a lier, " & compteurRouge & " en rouge)"
            DoEvents
        End If

        motBrut = Trim(wTexte(idxW))
        motCourant = RetirerPonctuationFinale(motBrut)
        aFusionne = False

        ' --- Tentative de fusion autour d'un tiret : "FRAG1" + "-" + "FRAG2" ---
        If EstUnSigle(motCourant) And idxW + 2 <= totalWords Then
            motSuivant1 = Trim(wTexte(idxW + 1))
            motSuivant2 = RetirerPonctuationFinale(Trim(wTexte(idxW + 2)))

            If motSuivant1 = "-" And EstUnSigle(motSuivant2) Then
                motCombine = motCourant & "-" & motSuivant2
                If dico.Exists(motCombine) Then
                    finCombine = wStart(idxW + 2) + Len(motSuivant2)
                    motCourant = motCombine
                    finPropre = finCombine
                    aFusionne = True
                End If
            End If
        End If

        ' [v116] Pluriel d'un sigle connu : "ONGs" -> sigle ONG, lien sur "ONGs"
        lenAffiche = Len(motCourant)
        If Not aFusionne And Len(motCourant) >= 3 Then
            If Right(motCourant, 1) = "s" Then
                motBase = Left(motCourant, Len(motCourant) - 1)
                If EstUnSigle(motBase) Then
                    If dico.Exists(motBase) Then motCourant = motBase
                End If
            End If
        End If

        If (EstUnSigle(motCourant) Or aFusionne) And Not dicoExceptions.Exists(motCourant) Then

            ' Un seul objet Range créé ICI, uniquement pour les
            Set wrd = ActiveDocument.Range(wStart(idxW), wFin(idxW))

            ' [v63] OPTIMISATION DE PERFORMANCE + comportement
            ' [v81] FACTORISATION : exclusion colonne via helper commun
            Dim dansTableauMot As Boolean
            dansTableauMot = EstDansColonneSigleExclue(wStart(idxW), wrd, tblDebuts, tblFins, tblColSigle, nbTables)

            If dico.Exists(motCourant) Then
                If comptesTrouve.Exists(motCourant) Then
                    comptesTrouve(motCourant) = comptesTrouve(motCourant) + 1
                Else
                    comptesTrouve.Add motCourant, 1
                End If
            End If

            ' [v46] La collection Words de Word inclut souvent
            If Not aFusionne Then
                finPropre = wStart(idxW) + lenAffiche
            End If

            If EstDansTitre(wrd) Then
                nbIgnoreTitre = nbIgnoreTitre + 1

            ElseIf wrd.Hyperlinks.Count > 0 Then
                nbIgnoreLien = nbIgnoreLien + 1

            ElseIf dansTableauMot Then
                nbIgnoreAutre = nbIgnoreAutre + 1

            ElseIf dico.Exists(motCourant) Then

                Dim contexte As String
                Dim posDebutCtx As Long
                Dim posFinCtx As Long
                contexte = ""
                On Error Resume Next
                posDebutCtx = wStart(idxW) - 15
                If posDebutCtx < 0 Then posDebutCtx = 0
                posFinCtx = finPropre + 15
                If posFinCtx > ActiveDocument.Content.End Then posFinCtx = ActiveDocument.Content.End
                contexte = ActiveDocument.Range(posDebutCtx, posFinCtx).Text
                contexte = Replace(contexte, vbCr, " ")
                contexte = Replace(contexte, vbLf, " ")
                On Error GoTo 0

                If nbAColler > UBound(debuts) Then
                    ReDim Preserve debuts(0 To nbAColler + 500)
                    ReDim Preserve fins(0 To nbAColler + 500)
                    ReDim Preserve sigleAssocie(0 To nbAColler + 500)
                    ReDim Preserve contextes(0 To nbAColler + 500)
                End If
                debuts(nbAColler) = wStart(idxW)
                fins(nbAColler) = finPropre
                sigleAssocie(nbAColler) = motCourant
                contextes(nbAColler) = contexte
                nbAColler = nbAColler + 1

            Else
                ' [v61] Le surlignage n'est plus appliqué
                If nbRouge > UBound(rougeDebuts) Then
                    ReDim Preserve rougeDebuts(0 To nbRouge + 500)
                    ReDim Preserve rougeFins(0 To nbRouge + 500)
                End If
                rougeDebuts(nbRouge) = wStart(idxW)
                rougeFins(nbRouge) = finPropre
                nbRouge = nbRouge + 1
                compteurRouge = compteurRouge + 1
            End If

        End If

        If aFusionne Then
            idxW = idxW + 3 ' saute les 2 mots suivants déjà fusionnés (tiret + fragment 2)
        Else
            idxW = idxW + 1
        End If

    Loop

    tClassif = Timer - tDebutPhase
    tDebutPhase = Timer

    ' =====================================================
    ' [v61] Application GROUPÉE de tous les surlignages rouges
    ' =====================================================
    Dim kRouge As Long
    For kRouge = 0 To nbRouge - 1
        On Error Resume Next
        ActiveDocument.Range(rougeDebuts(kRouge), rougeFins(kRouge)).HighlightColorIndex = wdRed
        On Error GoTo 0
        If kRouge Mod 200 = 0 Then
            Application.StatusBar = "Application du surlignage rouge : " & kRouge & "/" & nbRouge
            DoEvents
        End If
    Next kRouge

    Dim tRouge As Single
    tRouge = Timer - tDebutPhase
    tDebutPhase = Timer

    Dim k2 As Long
    Dim rngLien As Range
    Dim h As Hyperlink
    Dim tip As String
    Dim etaitRouge As Boolean

    For k2 = nbAColler - 1 To 0 Step -1

        If k2 Mod 100 = 0 Then
            Application.StatusBar = "Creation des liens : " & (nbAColler - k2) & "/" & nbAColler
            DoEvents
        End If

        tip = dico(sigleAssocie(k2))
        If Len(tip) > 80 Then tip = Left(tip, 80)

        Set rngLien = ActiveDocument.Range(debuts(k2), fins(k2))

        ' [v102] Sigle désormais connu mais encore surligné en rouge
        ' par un traitement précédent : on retire le rouge (il
        ' subsistait sous l'infobulle une fois le lien créé).
        etaitRouge = False
        On Error Resume Next
        etaitRouge = (rngLien.HighlightColorIndex = wdRed)
        If etaitRouge Then rngLien.HighlightColorIndex = wdNoHighlight
        On Error GoTo 0

        Dim eraitGras As Boolean
        Dim tailleOrigine As Variant
        eraitGras = False
        tailleOrigine = Empty
        On Error Resume Next
        eraitGras = (rngLien.Font.Bold = True)
        tailleOrigine = rngLien.Font.Size
        On Error GoTo 0

        On Error Resume Next
        Err.Clear
        ActiveDocument.Hyperlinks.Add _
            Anchor:=rngLien, _
            Address:=" ", _
            ScreenTip:=tip

        If Err.Number <> 0 Then
            rapportDetail = rapportDetail & sigleAssocie(k2) & " [" & debuts(k2) & "] ERREUR CREATION LIEN (" & Err.Number & " : " & Err.Description & ") - contexte: ..." & contextes(k2) & "..." & vbCrLf
            Debug.Print "Erreur creation lien [" & sigleAssocie(k2) & "] pos " & debuts(k2) & " : " & Err.Number & " - " & Err.Description
            Err.Clear
            If etaitRouge Then rngLien.HighlightColorIndex = wdRed
            On Error GoTo 0
            GoTo SuivantOccurrence
        End If
        On Error GoTo 0

        ' [v65] Propriétés redondantes retirées : ColorIndex,
        On Error Resume Next
        Err.Clear
        ZoneLien(rngLien).Style = ActiveDocument.Styles(wdStyleDefaultParagraphFont)
        Err.Clear
        ZoneLien(rngLien).Font.Bold = eraitGras
        Err.Clear
        If Not IsEmpty(tailleOrigine) Then
            ZoneLien(rngLien).Font.Size = tailleOrigine
            Err.Clear
        End If
        ZoneLien(rngLien).Font.Shading.Texture = wdTexture15Percent
        Err.Clear
        On Error GoTo 0

        ' [v125] Vérifie que le lien existe VRAIMENT (création sans erreur
        ' mais sans effet dans certaines cellules)
        If ActiveDocument.Range(debuts(k2), fins(k2)).Hyperlinks.Count = 0 And rngLien.Hyperlinks.Count = 0 Then
            nbLiensSansEffet = nbLiensSansEffet + 1
            Debug.Print "LIEN SANS EFFET [" & sigleAssocie(k2) & "] pos " & debuts(k2) & "-" & fins(k2) & " contexte: ..." & contextes(k2) & "..."
            GoTo SuivantOccurrence
        End If

        compteur = compteur + 1
        If comptesLie.Exists(sigleAssocie(k2)) Then
            comptesLie(sigleAssocie(k2)) = comptesLie(sigleAssocie(k2)) + 1
        Else
            comptesLie.Add sigleAssocie(k2), 1
        End If

SuivantOccurrence:
    Next k2

    Dim cleRapport As Variant
    Dim nbT As Long
    Dim nbL As Long
    For Each cleRapport In dico.Keys
        nbT = 0
        nbL = 0
        If comptesTrouve.Exists(cleRapport) Then nbT = comptesTrouve(cleRapport)
        If comptesLie.Exists(cleRapport) Then nbL = comptesLie(cleRapport)
        rapportDetail = rapportDetail & cleRapport & " : " & nbT & " trouve(s), " & nbL & " lie(s)"
        If nbT <> nbL Then rapportDetail = rapportDetail & "  <-- ECART"
        rapportDetail = rapportDetail & vbCrLf
    Next cleRapport

    tLiens = Timer - tDebutPhase

    Debug.Print "TraiterDocumentUnePasse : " & compteur & " lien(s), " & compteurRouge & " rouge(s), " & _
        nbIgnoreTitre & " ignore(s) titre/TOC, " & nbIgnoreLien & " ignore(s) deja lie, " & nbIgnoreAutre & " ignore(s) autre"

    ' [v60] Détail du chronométrage par phase

' [v108] Retire le surlignage ROUGE (et lui seul) sur une plage,
' caractère par caractère : fonctionne même si la plage est mixte
' (HighlightColorIndex = wdUndefined) ou dans une cellule de
' tableau. Retourne True si du rouge a été retiré.
Private Function RetirerRougeSurPlage(rng As Range) As Boolean
    Dim ch As Range
    Dim retire As Boolean
    retire = False
    If rng.HighlightColorIndex = wdNoHighlight Then Exit Function
    If rng.HighlightColorIndex = wdRed Then
        rng.HighlightColorIndex = wdNoHighlight
        RetirerRougeSurPlage = True
        Exit Function
    End If
    For Each ch In rng.Characters
        If ch.HighlightColorIndex = wdRed Then
            ch.HighlightColorIndex = wdNoHighlight
            retire = True
        End If
    Next ch
    RetirerRougeSurPlage = retire
End Function

Private Function RetirerPonctuationFinale(s As String) As String
    Dim t As String
    Dim c As String
    t = s
    Do While Len(t) > 0
        c = Right(t, 1)
        If (c >= "A" And c <= "Z") Or (c >= "0" And c <= "9") Then
            Exit Do
        End If
        t = Left(t, Len(t) - 1)
    Loop
    RetirerPonctuationFinale = t
End Function

' [v102] FACTORISATION : pour un sigle dont la définition vient
' d'être connue (ajout manuel, import du glossaire personnel),
' repère toutes ses occurrences actuellement surlignées en ROUGE
' (sigles inconnus lors d'un traitement précédent), retire le
' rouge et les transforme en lien avec infobulle (même rendu que
' le traitement principal). Retourne le nombre d'occurrences
' converties. Ne gère ni suivi des modifications ni Annuler : à
' l'appelant de les encadrer (une seule fois pour un lot).
' [v126] Traite UNE plage (cellule de définition) comme Traiter : lie les
' sigles connus (infobulle), surligne en rouge les sigles inconnus.
' Retourne le nombre de liens créés.
Private Function TraiterPlageSigles(rng As Range, dico As Object, dicoExc As Object) As Long
    Dim wS() As Long, wF() As Long, wT() As String, nW As Long
    Dim k As Long, i As Long
    Dim mB As String, mC As String, base As String
    Dim lenA As Long
    Dim wr As Range
    Dim d() As Long, f() As Long, sg() As String, rg() As Boolean
    Dim nb As Long
    Dim rngLien As Range
    Dim etaitGras As Boolean
    Dim etaitRouge As Boolean
    Dim tailleOrigine As Variant
    Dim tip As String
    Dim nbLiens As Long

    MaterialiserMots wS, wF, wT, nW, rng
    If nW = 0 Then Exit Function
    ReDim d(0 To nW)
    ReDim f(0 To nW)
    ReDim sg(0 To nW)
    ReDim rg(0 To nW)

    For k = 1 To nW
        mB = Trim(wT(k))
        mC = RetirerPonctuationFinale(mB)
        lenA = Len(mC)
        If Len(mC) >= 3 And Right(mC, 1) = "s" Then
            base = Left(mC, Len(mC) - 1)
            If EstUnSigle(base) Then
                If dico.Exists(base) Then mC = base
            End If
        End If
        If EstUnSigle(mC) Then
            If Not dicoExc.Exists(mC) Then
                Set wr = ActiveDocument.Range(wS(k), wS(k) + lenA)
                If wr.Hyperlinks.Count = 0 Then
                    d(nb) = wS(k)
                    f(nb) = wS(k) + lenA
                    sg(nb) = mC
                    rg(nb) = Not dico.Exists(mC)
                    nb = nb + 1
                End If
            End If
        End If
    Next k

    For i = nb - 1 To 0 Step -1
        Set rngLien = ActiveDocument.Range(d(i), f(i))
        If rg(i) Then
            On Error Resume Next
            rngLien.HighlightColorIndex = wdRed
            On Error GoTo 0
        Else
            tip = dico(sg(i))
            If Len(tip) > 80 Then tip = Left(tip, 80)
            etaitRouge = False
            On Error Resume Next
            etaitRouge = (rngLien.HighlightColorIndex = wdRed)
            On Error GoTo 0
            If etaitRouge Then rngLien.HighlightColorIndex = wdNoHighlight
            etaitGras = False
            tailleOrigine = Empty
            On Error Resume Next
            etaitGras = (rngLien.Font.Bold = True)
            tailleOrigine = rngLien.Font.Size
            Err.Clear
            ActiveDocument.Hyperlinks.Add Anchor:=rngLien, Address:=" ", ScreenTip:=tip
            If Err.Number = 0 Then
                ZoneLien(rngLien).Style = ActiveDocument.Styles(wdStyleDefaultParagraphFont)
                ZoneLien(rngLien).Font.Bold = etaitGras
                If Not IsEmpty(tailleOrigine) Then ZoneLien(rngLien).Font.Size = tailleOrigine
                ZoneLien(rngLien).Font.Shading.Texture = wdTexture15Percent
                nbLiens = nbLiens + 1
            Else
                If etaitRouge Then rngLien.HighlightColorIndex = wdRed
            End If
            Err.Clear
            On Error GoTo 0
        End If
    Next i

    TraiterPlageSigles = nbLiens
End Function

' [v126] Après ajout de lignes au tableau (import, définition, restauration) :
' traite la CELLULE DÉFINITION de chaque ligne ajoutée, pour que les
' sigles qu'elle cite soient liés (ou en rouge s'ils sont inconnus).
' "sigles" = dictionnaire dont les clés sont les sigles ajoutés.
Private Function TraiterDefinitionsAjoutees(sigles As Object) As Long
    Dim tbl As Table
    Dim posS As Long, posD As Long
    Dim tD() As Long, tF() As Long, tC() As Long, nbT As Long
    Dim dico As Object, dicoExc As Object
    Dim r As Long
    Dim txtS As String
    Dim n As Long
    Dim cel As Cell

    If sigles Is Nothing Then Set sigles = CreateObject("Scripting.Dictionary")
    If sigles.Count = 0 And Not tableauReecrit Then Exit Function

    Set tbl = TrouverTableauSiglesPourAjout(posS, posD)
    If tbl Is Nothing Then Exit Function
    Set dico = ConstruireDictionnaireComplet(tD, tF, tC, nbT)
    Set dicoExc = ChargerExceptions()

    For r = 2 To tbl.Rows.Count
        txtS = ""
        On Error Resume Next
        txtS = UCase(Trim(NettoyerTexteCellule(tbl.Cell(r, posS).Range.Text)))
        On Error GoTo 0
        If Len(txtS) > 0 Then
            If sigles.Exists(txtS) Or tableauReecrit Then
                ' [v140] définition ET colonnes de commentaires (toutes sauf la colonne sigle)
                Dim cC As Long
                For cC = tbl.Rows(r).Cells.Count To 1 Step -1
                    If cC <> posS Then
                        Set cel = Nothing
                        On Error Resume Next
                        Set cel = tbl.Cell(r, cC)
                        On Error GoTo 0
                        If Not cel Is Nothing Then n = n + TraiterPlageSigles(cel.Range, dico, dicoExc)
                    End If
                Next cC
            End If
        End If
    Next r
    TraiterDefinitionsAjoutees = n
    tableauReecrit = False
End Function

Private Function AppliquerDefinitionAuxOccurrencesRouges(sigle As String, definition As String) As Long

    ' [v124] Ne se limite plus aux occurrences ROUGES : lie TOUTES les
    ' occurrences du sigle (mot entier, casse respectée) qui ne sont pas
    ' encore des liens, hors titres/TOC et hors colonne sigle du tableau.
    ' = l'équivalent de "Traiter", mais ciblé sur ce seul sigle (recherche
    ' Find, sans relire tout le document). Utilisé après un import, un
    ' ajout de définition ou une restauration.

    ' --- Variables ---
    Dim posDebuts() As Long
    Dim posFins() As Long
    Dim nbTrouves As Long
    Dim rngRecherche As Range
    Dim nbAppliques As Long

    Dim rngLien As Range
    Dim etaitGras As Boolean
    Dim etaitRouge As Boolean

    Dim tip As String

    Dim posSignifRef As Long
    Dim dansTabRef As Boolean
    Dim exclure As Boolean
    Dim celOcc As Cell

    ' Un sigle en exception n'est jamais lié
    If ChargerExceptions().Exists(UCase(sigle)) Then
        AppliquerDefinitionAuxOccurrencesRouges = 0
        Exit Function
    End If

    ReDim posDebuts(0 To 200)
    ReDim posFins(0 To 200)
    nbTrouves = 0
    nbAppliques = 0

    ' Tableau de référence (pour exclure la colonne sigle)
    Set tblRef = TrouverTableauSiglesPourAjout(posSigleRef, posSignifRef)

    Set rngRecherche = ActiveDocument.Content

    Do While rngRecherche.Find.Execute(FindText:=sigle, MatchWholeWord:=True, MatchCase:=True, Forward:=True, Wrap:=wdFindStop, MatchWildcards:=False)

        exclure = False

        ' Déjà un lien
        If rngRecherche.Hyperlinks.Count > 0 Then exclure = True

        ' Titre / table des matières
        If Not exclure Then
            If EstDansTitre(rngRecherche) Then exclure = True
        End If

        ' Colonne sigle du tableau de référence
        If Not exclure And Not tblRef Is Nothing Then
            dansTabRef = False
            On Error Resume Next
            dansTabRef = rngRecherche.Information(wdWithInTable)
            If dansTabRef Then
                Set celOcc = Nothing
                Set celOcc = rngRecherche.Cells(1)
                If Not celOcc Is Nothing Then
                    If celOcc.Tables(1).Range.Start = tblRef.Range.Start Then
                        If celOcc.ColumnIndex = posSigleRef Then exclure = True
                        If celOcc.RowIndex = 1 Then exclure = True
                    End If
                End If
            End If
            On Error GoTo 0
        End If

        If Not exclure Then
            If nbTrouves > UBound(posDebuts) Then
                ReDim Preserve posDebuts(0 To nbTrouves + 200)
                ReDim Preserve posFins(0 To nbTrouves + 200)
            End If
            posDebuts(nbTrouves) = rngRecherche.Start
            posFins(nbTrouves) = rngRecherche.End
            nbTrouves = nbTrouves + 1
        End If

        rngRecherche.Collapse wdCollapseEnd
        If nbTrouves > 5000 Then Exit Do

    Loop

    tip = definition
    If Len(tip) > 80 Then tip = Left(tip, 80)

    For k = nbTrouves - 1 To 0 Step -1

        Set rngLien = ActiveDocument.Range(posDebuts(k), posFins(k))

        etaitRouge = False
        On Error Resume Next
        etaitRouge = (rngLien.HighlightColorIndex = wdRed)
        On Error GoTo 0
        If etaitRouge Then rngLien.HighlightColorIndex = wdNoHighlight

        etaitGras = False
        tailleOrigine = Empty
        On Error Resume Next
        etaitGras = (rngLien.Font.Bold = True)
        tailleOrigine = rngLien.Font.Size
        On Error GoTo 0

        On Error Resume Next
        Err.Clear
        ActiveDocument.Hyperlinks.Add Anchor:=rngLien, Address:=" ", ScreenTip:=tip
        If Err.Number = 0 Then
            ZoneLien(rngLien).Style = ActiveDocument.Styles(wdStyleDefaultParagraphFont)
            ZoneLien(rngLien).Font.Bold = etaitGras
            If Not IsEmpty(tailleOrigine) Then ZoneLien(rngLien).Font.Size = tailleOrigine
            ZoneLien(rngLien).Font.Shading.Texture = wdTexture15Percent
            nbAppliques = nbAppliques + 1
        Else
            If etaitRouge Then rngLien.HighlightColorIndex = wdRed ' échec : on remet le rouge
        End If
        Err.Clear
        On Error GoTo 0

    Next k

    AppliquerDefinitionAuxOccurrencesRouges = nbAppliques

End Function

Private Function TrouverTableauSiglesPourAjout(ByRef posSigleOut As Long, ByRef posSignifOut As Long) As Table
    Dim tbl As Table
    Dim cel As Cell
    Dim texteEntete As String
    Dim colReelle As Long

    posSigleOut = 0
    posSignifOut = 0

    ' [v76] Récupère les intitulés personnalisés éventuellement
    Dim intituleSigleP As String
    Dim intituleSignifP As String
    On Error Resume Next
    intituleSigleP = ActiveDocument.Variables(NOM_VARIABLE_INTITULE_SIGLE).Value
    intituleSignifP = ActiveDocument.Variables(NOM_VARIABLE_INTITULE_SIGNIF).Value
    On Error GoTo 0

    For Each tbl In ActiveDocument.Tables

        Dim pSigle As Long
        Dim pSignif As Long
        pSigle = 0
        pSignif = 0

        On Error Resume Next
        For Each cel In tbl.Rows(1).Cells
            colReelle = cel.ColumnIndex
            texteEntete = NettoyerTexteCellule(cel.Range.Text)

            If pSigle = 0 Then
                If ContientMotCle(texteEntete, Array("sigle", "acronyme", "abreviation", "abréviation", "code", "trigramme", "raccourci", "terme", "mot")) Then

' [v75] Ajoute une nouvelle ligne à LA FIN du premier tableau de

' [v111] TRI MAISON d'un tableau de sigles (remplace Table.Sort
' natif qui dupliquait le tableau/le titre). Accepte une ligne de
' titre à cellules FUSIONNÉES : seules les lignes 2..n sont lues
' (via Rows(r).Cells(c), pas tbl.Cell ni tbl.Columns qui échouent
' sur un tableau non uniforme). Tri stable sur la colonne du sigle,
' insensible à la casse. Supprime un titre répété dans le corps et
' les lignes strictement identiques consécutives, puis ne réécrit
' QUE les cellules modifiées. Retourne le nombre de lignes
' supprimées, ou -1 si le tri est impossible (cause dans
' derniereErreurTri). N'appelle JAMAIS Table.Sort.
Private Function TrierTableauManuel(tbl As Table, colS As Long) As Long

    Dim nL As Long, nC As Long
    Dim r As Long, c As Long, i As Long, j As Long, m As Long, k As Long
    Dim cel() As String
    Dim ordre() As Long
    Dim cle As Long
    Dim identique As Boolean
    Dim txtTitre As String
    Dim txtLigne As String

    derniereErreurTri = ""
    On Error GoTo Echec

    nL = tbl.Rows.Count
    If nL < 3 Then
        TrierTableauManuel = 0
        Exit Function
    End If
    nC = tbl.Rows(2).Cells.Count
    If colS < 1 Or colS > nC Then
        derniereErreurTri = "colonne du sigle hors tableau"
        GoTo Echec
    End If

    ' Lecture lignes 2..n (toutes doivent avoir le même nombre de cellules)
    ReDim cel(2 To nL, 1 To nC)
    For r = 2 To nL
        If tbl.Rows(r).Cells.Count <> nC Then
            derniereErreurTri = "la ligne " & r & " n'a pas le même nombre de cellules (cellules fusionnées dans le corps du tableau)"
            GoTo Echec
        End If
        For c = 1 To nC
            cel(r, c) = tbl.Rows(r).Cells(c).Range.Text
            If Len(cel(r, c)) >= 2 Then cel(r, c) = Left(cel(r, c), Len(cel(r, c)) - 2)
        Next c
    Next r

    ' Texte de la ligne de titre : globalement, et pour la cellule de
    ' la colonne du sigle (ColumnIndex réel, titre fusionné toléré)
    txtTitre = Trim(tbl.Rows(1).Range.Text)
    Dim txtTitreSigle As String
    Dim ceT As Cell
    txtTitreSigle = ""
    On Error Resume Next
    For Each ceT In tbl.Rows(1).Cells
        If ceT.ColumnIndex = colS Then
            txtTitreSigle = Trim(Left(ceT.Range.Text, Len(ceT.Range.Text) - 2))
            Exit For
        End If
    Next ceT
    On Error GoTo Echec

    ' Lignes conservées : on écarte les lignes VIDES, les titres
    ' répétés (ligne entière ou cellule sigle = titre de colonne)
    Dim ecarter As Boolean
    ReDim ordre(1 To nL)
    m = 0
    For r = 2 To nL
        txtLigne = Trim(tbl.Rows(r).Range.Text)
        ecarter = False
        If StrComp(txtLigne, txtTitre, vbTextCompare) = 0 Then ecarter = True
        If Not ecarter And Len(txtTitreSigle) > 0 Then
            If StrComp(Trim(cel(r, colS)), txtTitreSigle, vbTextCompare) = 0 Then ecarter = True
        End If
        If Not ecarter Then
            ecarter = True
            For c = 1 To nC
                If Len(Trim(cel(r, c))) > 0 Then ecarter = False: Exit For
            Next c
        End If
        If Not ecarter Then
            m = m + 1
            ordre(m) = r
        End If
    Next r

    ' Tri par insertion, stable
    For i = 2 To m
        cle = ordre(i)
        j = i - 1
        Do While j >= 1
            If StrComp(Trim(cel(ordre(j), colS)), Trim(cel(cle, colS)), vbTextCompare) <= 0 Then Exit Do
            ordre(j + 1) = ordre(j)
            j = j - 1
        Loop
        ordre(j + 1) = cle
    Next i

    ' Suppression des lignes strictement identiques consécutives
    k = 0
    For i = 1 To m
        identique = False
        If k >= 1 Then
            identique = True
            For c = 1 To nC
                If StrComp(Trim(cel(ordre(k), c)), Trim(cel(ordre(i), c)), vbTextCompare) <> 0 Then identique = False: Exit For
            Next c
        End If
        If Not identique Then
            k = k + 1
            ordre(k) = ordre(i)
        End If
    Next i
    m = k

    ' Réécriture des seules cellules modifiées (lignes 2..m+1)
    For i = 1 To m
        For c = 1 To nC
            If cel(i + 1, c) <> cel(ordre(i), c) Then
                tbl.Rows(i + 1).Cells(c).Range.Text = cel(ordre(i), c)
                ' [v134] le texte réécrit héritait du surlignage rouge / ombrage de l'ancien contenu
                tbl.Rows(i + 1).Cells(c).Range.HighlightColorIndex = wdNoHighlight
                tbl.Rows(i + 1).Cells(c).Range.Font.Shading.Texture = wdTextureNone
                tableauReecrit = True
            End If
        Next c
    Next i

    ' Suppression des lignes en trop (en partant de la fin)
    For r = nL To m + 2 Step -1
        tbl.Rows(r).Delete
    Next r

    TrierTableauManuel = nL - 1 - m
    Exit Function

Echec:
    If Len(derniereErreurTri) = 0 Then derniereErreurTri = "erreur " & Err.Number & " : " & Err.Description
    TrierTableauManuel = -1
End Function

' [v119] DIAGNOSTIC : sélectionnez UN mot (ex. TESAR dans la définition

' DiagnostiquerMot). Affiche le découpage Word, les codes des caractères
' et chaque contrôle du traitement, pour voir lequel écarte le mot.
Private Function CodesCar(s As String) As String
    Dim i As Long
    Dim r As String
    For i = 1 To Len(s)
        If i > 25 Then
            r = r & "..."
            Exit For
        End If
        r = r & Mid(s, i, 1) & "[" & AscW(Mid(s, i, 1)) & "] "
    Next i
    CodesCar = r
End Function

Sub DiagnostiquerMot()
    Dim rng As Range
    Dim rw As Range
    Dim rn As Range
    Dim rp As Range
    Dim msg As String
    Dim motNet As String
    Dim tblD() As Long, tblF() As Long, tblC() As Long, nbT As Long
    Dim dico As Object
    Dim dicoExc As Object
    Dim ce As Cell
    Dim dansTab As Boolean
    Dim colAct As Long

    Set rn = rw.Next(wdWord, 1)
    If Not rp Is Nothing Then msg = msg & "Mot avant  : " & CodesCar(rp.Text) & vbCrLf
    If Not rn Is Nothing Then msg = msg & "Mot après  : " & CodesCar(rn.Text) & vbCrLf

    msg = msg & vbCrLf & "Mot nettoyé : [" & motNet & "]" & vbCrLf

    msg = msg & "Dans le dictionnaire (tableau) : " & dico.Exists(motNet) & vbCrLf
    msg = msg & "Dans les exceptions : " & dicoExc.Exists(motNet) & vbCrLf
    msg = msg & "Déjà un lien : " & (rw.Hyperlinks.Count > 0) & vbCrLf
    msg = msg & "Titre / TOC : " & EstDansTitre(rw) & vbCrLf

    dansTab = False
    On Error Resume Next
    dansTab = rw.Information(wdWithInTable)
    On Error GoTo 0
    msg = msg & "Dans un tableau : " & dansTab & vbCrLf
    If dansTab Then
        colAct = -1
        On Error Resume Next
        colAct = rw.Cells(1).ColumnIndex
        On Error GoTo 0
        msg = msg & "  colonne=" & colAct & vbCrLf
        msg = msg & "  exclu (colonne sigle) : " & EstDansColonneSigleExclue(rw.Start, rw, tblD, tblF, tblC, nbT) & vbCrLf
    End If

    ' [v122] SIMULATION de la boucle de Traiter sur ce mot : mêmes tableaux
    ' (MaterialiserMots), mêmes contrôles, dans le même ordre.
    Dim sS() As Long, sF() As Long, sT() As String, sN As Long
    Dim k As Long, idxS As Long, kk As Long
    Dim mC As String, wS As Range
    MaterialiserMots sS, sF, sT, sN
    idxS = 0
    For k = 1 To sN
        If rng.Start >= sS(k) And rng.Start < sF(k) Then
            idxS = k
            Exit For
        End If
    Next k
    msg = msg & vbCrLf & "--- SIMULATION Traiter ---" & vbCrLf

    If idxS = 0 Then
        msg = msg & "AUCUN mot matérialisé ne contient cette position : le mot est PERDU dans MaterialiserMots." & vbCrLf
    Else
        For kk = idxS - 1 To idxS + 2
            If kk >= 1 And kk <= sN Then
                msg = msg & IIf(kk = idxS, ">> ", "   ") & "mot " & kk & " [" & sS(kk) & "-" & sF(kk) & "] " & CodesCar(sT(kk)) & vbCrLf
            End If
        Next kk
        mC = RetirerPonctuationFinale(Trim(sT(idxS)))
        msg = msg & "motCourant = [" & mC & "] EstUnSigle=" & EstUnSigle(mC) & vbCrLf

        msg = msg & "Plage = [" & wS.Text & "] len=" & Len(wS.Text) & vbCrLf
        msg = msg & "  exception=" & dicoExc.Exists(mC) & " dico=" & dico.Exists(mC) & vbCrLf

        If idxS + 2 <= sN Then
            msg = msg & "  fusion tiret testée avec: [" & Trim(sT(idxS + 1)) & "]" & vbCrLf
        End If
    End If

    ' [v123] ESSAI de création du lien sur ce mot, avec les mêmes appels que
    ' Traiter, et rapport de l'erreur éventuelle.
    If idxS > 0 Then
        If dico.Exists(mC) And wS.Hyperlinks.Count = 0 Then
            If MsgBox("Tenter de créer maintenant le lien sur [" & mC & "] (essai, Ctrl+Z pour annuler) ?", vbYesNo + vbQuestion, "Essai de création") = vbYes Then
                Dim rngEssai As Range
                Dim tipEssai As String
                tipEssai = dico(mC)
                If Len(tipEssai) > 80 Then tipEssai = Left(tipEssai, 80)
                Set rngEssai = ActiveDocument.Range(sS(idxS), sS(idxS) + Len(mC))

                msg = msg & "Plage essai = [" & rngEssai.Text & "] " & rngEssai.Start & "-" & rngEssai.End & vbCrLf

                Err.Clear
                ActiveDocument.Hyperlinks.Add Anchor:=rngEssai, Address:=" ", ScreenTip:=tipEssai
                msg = msg & "Erreur n° " & Err.Number & " : " & Err.Description & vbCrLf
                Err.Clear
                On Error GoTo 0
                msg = msg & "Liens sur la plage après essai : " & ActiveDocument.Range(sS(idxS), sS(idxS) + Len(mC)).Hyperlinks.Count & vbCrLf
            End If
        End If
    End If

    ' [v120] Résultat dans un NOUVEAU document (copiable), pas dans une MsgBox
    Debug.Print msg
    Dim docDiag As Document
    Set docDiag = Documents.Add
    docDiag.Content.Text = "DIAGNOSTIC DU MOT" & vbCr & Replace(msg, vbCrLf, vbCr)
    docDiag.Activate
End Sub

' [v121] RESTAURATION : retrouve les sigles dont la ligne a été supprimée
' du tableau (ancien bug de "Supprimer sigles inutilisés") alors que
' leurs LIENS avec infobulle sont restés dans le document. L'infobulle
' contient la définition : on la réinjecte dans le tableau.
Private Function EstSigleOuTiret(s As String) As Boolean
    Dim i As Long
    Dim c As String
    Dim aLettre As Boolean
    EstSigleOuTiret = False
    If Len(s) < 2 Or Len(s) > 15 Then Exit Function
    For i = 1 To Len(s)
        c = Mid(s, i, 1)
        If c >= "A" And c <= "Z" Then
            aLettre = True
        ElseIf Not ((c >= "0" And c <= "9") Or c = "-") Then
            Exit Function
        End If
    Next i
    EstSigleOuTiret = aLettre
End Function

Sub RestaurerSiglesDepuisInfobulles()

    Dim tblD() As Long, tblF() As Long, tblC() As Long, nbT As Long
    Dim dico As Object
    Dim dicoExc As Object
    Dim manquants As Object
    Dim h As Hyperlink
    Dim i As Long, n As Long
    Dim txt As String, tip As String, base As String
    Dim cle As Variant
    Dim liste As String
    Dim nbAjoutes As Long
    Dim etatOpt As EtatOptionsWord

    Set dico = ConstruireDictionnaireComplet(tblD, tblF, tblC, nbT)
    If nbT = 0 Then
        MsgBox "Aucun tableau de sigles reconnu : restauration impossible.", vbExclamation
        Exit Sub
    End If
    Set dicoExc = ChargerExceptions()
    Set manquants = CreateObject("Scripting.Dictionary")
    manquants.CompareMode = vbTextCompare

    n = ActiveDocument.Hyperlinks.Count
    For i = 1 To n
        txt = ""
        tip = ""
        On Error Resume Next
        Set h = ActiveDocument.Hyperlinks(i)
        txt = h.TextToDisplay
        tip = h.ScreenTip
        On Error GoTo 0

        txt = Trim(RetirerPonctuationFinale(Trim(txt)))
        ' pluriel : "ONGs" -> "ONG"
        If Len(txt) >= 3 And Right(txt, 1) = "s" Then
            base = Left(txt, Len(txt) - 1)
            If EstSigleOuTiret(base) Then txt = base
        End If
        txt = UCase(txt)

        If EstSigleOuTiret(txt) And Len(Trim(tip)) >= 2 Then
            If Not dico.Exists(txt) And Not dicoExc.Exists(txt) And Not manquants.Exists(txt) Then
                manquants.Add txt, Trim(tip)
            End If
        End If

        If i Mod 50 = 0 Then
            Application.StatusBar = "Analyse des liens : " & i & "/" & n
            DoEvents
        End If
    Next i
    Application.StatusBar = False

    If manquants.Count = 0 Then
        MsgBox "Aucun lien orphelin : tous les sigles liés sont déjà dans le tableau.", vbInformation
        Exit Sub
    End If

    For Each cle In manquants.Keys
        liste = liste & cle & " : " & Left(manquants(cle), 50) & vbCrLf
        If Len(liste) > 1200 Then
            liste = liste & "..." & vbCrLf
            Exit For
        End If
    Next cle

    If MsgBox(manquants.Count & " sigle(s) lié(s) dans le texte mais absent(s) du tableau :" & vbCrLf & vbCrLf & liste & vbCrLf & _
              "Les ajouter au tableau (définition = infobulle) ?", vbYesNo + vbQuestion, "Restauration") = vbNo Then Exit Sub

    etatOpt = DesactiverOptionsTraitement()
    On Error Resume Next
    Application.UndoRecord.StartCustomRecord "Restauration des sigles"
    On Error GoTo 0

    For Each cle In manquants.Keys
        If AjouterLigneAuTableau(CStr(cle), CStr(manquants(cle))) Then
            nbAjoutes = nbAjoutes + 1
            AppliquerDefinitionAuxOccurrencesRouges CStr(cle), CStr(manquants(cle))
        End If
    Next cle

    TraiterDefinitionsAjoutees manquants

    On Error Resume Next
    Application.UndoRecord.EndCustomRecord
    On Error GoTo 0
    RestaurerOptionsTraitement etatOpt

    MsgBox nbAjoutes & " sigle(s) ajouté(s) au tableau (tri alphabétique)." & vbCrLf & _
           "Pensez à enregistrer le document (Ctrl+S).", vbInformation, "Restauration terminée"
End Sub

Private Function AjouterLigneAuTableau(sigle As String, definition As String) As Boolean
    Dim posSigleT As Long
    Dim posSignifT As Long
    Dim tbl As Table

    Set tbl = TrouverTableauSiglesPourAjout(posSigleT, posSignifT)

    If tbl Is Nothing Then
        AjouterLigneAuTableau = False
        Exit Function
    End If

    ' [v96] Avant toute insertion, on s'assure que le tableau est
    ' DÉJÀ correctement trié par ordre alphabétique (tri natif
    ' Table.Sort, silencieux en cas d'échec). Ceci couvre le cas
    ' d'un tableau non trié à l'origine ou déréglé par des ajouts/
    ' suppressions manuels dans Word : sans ce tri préalable, la
    ' recherche de position ci-dessous (qui suppose un tableau déjà
    ' ordonné) pourrait insérer la nouvelle ligne au mauvais endroit.
    ' [v111] Tri maison uniquement (Table.Sort natif supprimé : il
    ' dupliquait le tableau). Si le tri est impossible, on insère
    ' quand même, à la position calculée ci-dessous.
    TrierTableauManuel tbl, posSigleT

    ' [v94] Insertion directement à la bonne position ALPHABÉTIQUE
    ' au lieu de toujours ajouter en fin de tableau : parcourt les
    ' lignes existantes et insère juste avant la première dont le
    ' sigle est alphabétiquement supérieur au nouveau. Si aucune
    ' (le nouveau sigle est le "plus grand"), ajoute en fin comme
    ' avant.
    Dim rLig As Long
    Dim texteExistant As String
    Dim rowCible As Row
    Set rowCible = Nothing

    On Error Resume Next
    For rLig = 2 To tbl.Rows.Count
        texteExistant = NettoyerTexteCellule(tbl.Cell(rLig, posSigleT).Range.Text)
        If StrComp(texteExistant, sigle, vbTextCompare) > 0 Then
            Set rowCible = tbl.Rows(rLig)
            Exit For
        End If
    Next rLig
    On Error GoTo 0

    On Error Resume Next
    Dim nouvelleLigne As Row
    If rowCible Is Nothing Then
        Set nouvelleLigne = tbl.Rows.Add ' fin de tableau
    Else
        Set nouvelleLigne = tbl.Rows.Add(BeforeRow:=rowCible) ' position alphabétique
    End If

    nouvelleLigne.Cells(posSigleT).Range.Text = sigle
    nouvelleLigne.Cells(posSignifT).Range.Text = definition

    AjouterLigneAuTableau = (Err.Number = 0)
    Err.Clear
    On Error GoTo 0
End Function

' [v100] Bouton ruban "Configurer" : point d'entrée UNIQUE pour
' associer un glossaire personnel à la session Word en cours,
' que ce glossaire soit tout nouveau (premier lancement du
' projet) ou déjà existant. Totalement DÉCOUPLÉ du tableau du
' document : configurer ou synchroniser un glossaire ne modifie
' jamais silencieusement le tableau, et inversement modifier le
' tableau ne touche plus jamais le glossaire tant que le bouton
' "Synchroniser" n'est pas actionné explicitement (v99 faisait
' cette proposition automatiquement à chaque ajout - abandonné
' pour permettre de garder des documents "brouillon" ou
' spécifiques sans polluer le glossaire personnel).
Sub ConfigurerGlossairePersonnel()

    ' --- Variables ---
    Dim rep As VbMsgBoxResult
    Dim cheminCreation As String
    Dim numCreation As Integer

    rep = MsgBox( _
        "Configurer mon glossaire personnel" & vbCrLf & vbCrLf & _
        "Ce fichier externe (.txt) permet de réutiliser les mêmes sigles d'un document Word à l'autre, indépendamment du tableau de ce document." & vbCrLf & vbCrLf & _
        "Si vous liez un glossaire existant, ses sigles absents du tableau de ce document y sont ajoutés tout de suite, et leurs occurrences encore en rouge dans le texte deviennent des infobulles (le tableau reste prioritaire en cas de définition différente)." & vbCrLf & _
        "Le sens inverse (tableau vers glossaire) n'est jamais automatique : utilisez le bouton ""Synchroniser"" quand vous le souhaitez." & vbCrLf & vbCrLf & _
        "Oui = créer un nouveau glossaire (vide)" & vbCrLf & _
        "Non = lier un glossaire déjà existant" & vbCrLf & _
        "Annuler = ne rien faire", _
        vbYesNoCancel + vbQuestion, "Glossaire personnel")

    Select Case rep

        Case vbYes ' Créer un nouveau glossaire vide
            With Application.FileDialog(msoFileDialogSaveAs)
                .Title = "Créer le glossaire personnel"
                .InitialFileName = "MonGlossairePersonnel.txt"
                If .Show <> -1 Then Exit Sub
                cheminCreation = .SelectedItems(1)
            End With
            If LCase(Right(cheminCreation, 4)) <> ".txt" Then cheminCreation = cheminCreation & ".txt"

            On Error GoTo ErreurConfigGloss
            numCreation = FreeFile
            Open cheminCreation For Output As #numCreation ' crée/vide le fichier
            Close #numCreation
            On Error GoTo 0

            cheminGlossairePersoSession = cheminCreation
            MsgBox "Glossaire personnel créé et lié pour cette session :" & vbCrLf & cheminGlossairePersoSession, vbInformation

        Case vbNo ' Lier un glossaire existant
            With Application.FileDialog(msoFileDialogFilePicker)
                .Title = "Choisir le glossaire personnel à lier"
                .Filters.Clear
                .Filters.Add "Glossaire texte", "*.txt"
                .AllowMultiSelect = False
                If .Show <> -1 Then Exit Sub
                cheminGlossairePersoSession = .SelectedItems(1)
            End With
            MsgBox "Glossaire personnel lié pour cette session :" & vbCrLf & cheminGlossairePersoSession, vbInformation
            ' [v101] Lier = aussi importer (perso -> document)
            ImporterDepuisFichierGlossaire cheminGlossairePersoSession

        Case Else ' Annuler
            Exit Sub

    End Select

    Exit Sub

ErreurConfigGloss:
    On Error Resume Next
    Close #numCreation
    On Error GoTo 0
    MsgBox "Erreur lors de la création du glossaire personnel : " & Err.Description, vbExclamation

End Sub

' [v100] Bouton ruban "Synchroniser" : répercute EXPLICITEMENT,
' à la demande, les entrées du tableau du document actif vers le
' glossaire personnel lié (bouton "Configurer" requis avant).
' N'écrase jamais une entrée déjà présente dans le fichier ;
' n'ajoute que les sigles absents. Le tableau du document reste
' la seule source lue ici, jamais modifié en retour.
Sub SynchroniserGlossairePersonnel()

    ' --- Variables ---
    Dim tblDebutsSync() As Long
    Dim tblFinsSync() As Long
    Dim tblColSigleSync() As Long
    Dim nbTablesSync As Long
    Dim dicoTableau As Object
    Dim dicoFichierExistant As Object
    Dim numLectureManuel As Integer
    Dim ligneManuel As String
    Dim posBarreManuel As Long
    Dim sigleManuel As String
    Dim nbAjoutesSync As Long
    Dim numAjoutManuel As Integer
    Dim cleSync As Variant

    If cheminGlossairePersoSession = "" Then
        MsgBox "Aucun glossaire personnel lié pour cette session." & vbCrLf & _
               "Utilisez d'abord le bouton ""Configurer"".", vbExclamation
        Exit Sub
    End If

    Set dicoTableau = ConstruireDictionnaireComplet(tblDebutsSync, tblFinsSync, tblColSigleSync, nbTablesSync)

    If dicoTableau.Count = 0 Then
        MsgBox "Aucun sigle dans le tableau de ce document à synchroniser.", vbInformation
        Exit Sub
    End If

    ' Sigles déjà présents dans le fichier, pour ne pas les
    ' dupliquer ni les écraser.
    Set dicoFichierExistant = CreateObject("Scripting.Dictionary")
    dicoFichierExistant.CompareMode = vbTextCompare

    On Error GoTo ErreurSyncManuel
    If Len(Dir(cheminGlossairePersoSession)) > 0 Then
        numLectureManuel = FreeFile
        Open cheminGlossairePersoSession For Input As #numLectureManuel
        Do While Not EOF(numLectureManuel)
            Line Input #numLectureManuel, ligneManuel
            posBarreManuel = InStr(ligneManuel, "|")
            If posBarreManuel > 1 Then
                sigleManuel = Trim(Left(ligneManuel, posBarreManuel - 1))
                If Len(sigleManuel) > 0 And Not dicoFichierExistant.Exists(sigleManuel) Then
                    dicoFichierExistant.Add sigleManuel, Trim(Mid(ligneManuel, posBarreManuel + 1))
                End If
            End If
        Loop
        Close #numLectureManuel
    End If

    nbAjoutesSync = 0
    numAjoutManuel = FreeFile
    Open cheminGlossairePersoSession For Append As #numAjoutManuel
    For Each cleSync In dicoTableau.Keys
        If Not dicoFichierExistant.Exists(cleSync) Then
            Print #numAjoutManuel, cleSync & "|" & dicoTableau(cleSync)
            nbAjoutesSync = nbAjoutesSync + 1
        End If
    Next cleSync
    Close #numAjoutManuel
    On Error GoTo 0

    MsgBox nbAjoutesSync & " sigle(s) ajouté(s) au glossaire personnel." & vbCrLf & _
           (dicoTableau.Count - nbAjoutesSync) & " déjà présent(s), inchangé(s).", vbInformation, "Synchronisation terminée"
    Exit Sub

ErreurSyncManuel:
    On Error Resume Next
    Close #numLectureManuel
    Close #numAjoutManuel
    On Error GoTo 0
    MsgBox "Erreur pendant la synchronisation du glossaire personnel : " & Err.Description, vbExclamation

End Sub

' [v94] Trie (ou retrie) un tableau de sigles reconnu par ordre
' alphabétique sur sa colonne "sigle" — utile pour un tableau déjà
' existant non trié, ou après plusieurs ajouts manuels successifs.
Sub TrierTableauSigles()

    Dim tblDebuts() As Long
    Dim tblFins() As Long
    Dim tblColSigle() As Long
    Dim nbTablesSigle As Long
    Dim dico As Object
    Set dico = ConstruireDictionnaireComplet(tblDebuts, tblFins, tblColSigle, nbTablesSigle)

    If nbTablesSigle = 0 Then
        MsgBox "Aucun tableau de sigles reconnu à trier.", vbInformation
        Exit Sub
    End If

    Dim etatOpt As EtatOptionsWord
    etatOpt = DesactiverOptionsTraitement()

    Dim nbTriees As Long
    nbTriees = 0
    Dim tIdxTri As Long
    Dim nbDoublonsSupp As Long
    nbDoublonsSupp = 0

    For tIdxTri = 0 To nbTablesSigle - 1

        Dim tblTri As Table
        Set tblTri = Nothing
        Dim tblCand As Table
        On Error Resume Next
        For Each tblCand In ActiveDocument.Tables
            If tblCand.Range.Start = tblDebuts(tIdxTri) Then
                Set tblTri = tblCand
                Exit For
            End If
        Next tblCand
        On Error GoTo 0

        If Not tblTri Is Nothing Then
            ' [v110] Tri maison (Table.Sort natif dupliquait le titre)
            Dim retTri As Long
            retTri = TrierTableauManuel(tblTri, tblColSigle(tIdxTri))
            If retTri >= 0 Then
                nbTriees = nbTriees + 1
                nbDoublonsSupp = nbDoublonsSupp + retTri
            Else
                MsgBox "Tableau n° " & (tIdxTri + 1) & " non trié : " & derniereErreurTri, vbExclamation
            End If
        End If

    Next tIdxTri

    If tableauReecrit Then TraiterDefinitionsAjoutees Nothing
    RestaurerOptionsTraitement etatOpt

    MsgBox nbTriees & " tableau(x) de sigles trié(s) par ordre alphabétique." & vbCrLf & _
           nbDoublonsSupp & " ligne(s) en double supprimée(s) (lignes vides, titre répété ou lignes identiques)." & vbCrLf & _
           "Pensez à enregistrer le document (Ctrl+S).", vbInformation, "Terminé"

End Sub

' [v139] Désigne le tableau de la sélection comme tableau de sigles :
' mémorise les intitulés des 2 colonnes (variables du document).
Private Function DesignerTableauDepuisSelection() As Boolean
    Dim tb As Table
    Dim cMin As Long, cMax As Long
    Dim cel As Cell
    Dim intSigle As String, intDefin As String
    On Error GoTo Echec
    Set tb = Selection.Tables(1)
    cMin = 9999: cMax = 0
    For Each cel In Selection.Cells
        If cel.ColumnIndex < cMin Then cMin = cel.ColumnIndex
    Next cel
    ' [v140] Sigle = colonne la plus à gauche ; Définition = colonne
    ' IMMÉDIATEMENT à droite ; les suivantes sont des commentaires.
    cMax = cMin + 1
    intSigle = ""
    intDefin = ""
    For Each cel In tb.Rows(1).Cells
        If cel.ColumnIndex = cMin Then intSigle = NettoyerTexteCellule(cel.Range.Text)
        If cel.ColumnIndex = cMax Then intDefin = NettoyerTexteCellule(cel.Range.Text)
    Next cel
    If Len(intSigle) = 0 Or Len(intDefin) = 0 Or StrComp(intSigle, intDefin, vbTextCompare) = 0 Then GoTo Echec
    On Error Resume Next
    ActiveDocument.Variables(NOM_VARIABLE_INTITULE_SIGLE).Delete
    ActiveDocument.Variables(NOM_VARIABLE_INTITULE_SIGNIF).Delete
    On Error GoTo Echec
    ActiveDocument.Variables.Add NOM_VARIABLE_INTITULE_SIGLE, intSigle
    ActiveDocument.Variables.Add NOM_VARIABLE_INTITULE_SIGNIF, intDefin
    DesignerTableauDepuisSelection = True
    Exit Function
Echec:
    DesignerTableauDepuisSelection = False
End Function

' [v138] Crée, à la fin du document, un tableau "Sigle | Définition" vide

Echec:

' [v98] Importe un glossaire personnel externe (fichier .txt au
' format "SIGLE|Definition") et ajoute au tableau de sigles du
' document toute entrée dont le sigle n'y figure pas déjà. Le
' tableau reste AUTORITAIRE (cohérent avec la règle v93) : un
' sigle déjà présent dans le tableau n'est jamais modifié, quelle
' que soit sa définition dans le fichier importé.
' [v101] Devenue routine interne : le chemin est fourni par
' ConfigurerGlossairePersonnel (qui lie ET importe), plus de
' boîte de dialogue ni de bouton ruban dédié ici.
Private Sub ImporterDepuisFichierGlossaire(cheminFichier As String)

    ' --- Variables ---
    Dim dicoImport As Object
    Dim numFichier As Integer
    Dim ligneFichier As String
    Dim posSep As Long
    Dim sigleLu As String
    Dim defLue As String
    Dim posSigleT As Long
    Dim posSignifT As Long
    Dim tbl As Table
    Dim dicoExistant As Object
    Dim rLigneImp As Long
    Dim texteExistantImp As String
    Dim ancienSuiviModifImp As Boolean
    Dim cleImport As Variant
    Dim nbImportes As Long
    Dim nbIgnores As Long
    Dim nbRougesConverties As Long
    Dim dicoAjoutesImp As Object
    Dim nbLiensDefs As Long
    Set dicoAjoutesImp = CreateObject("Scripting.Dictionary")
    dicoAjoutesImp.CompareMode = vbTextCompare

    If Len(cheminFichier) = 0 Then Exit Sub

    ' --- Lecture du fichier et construction du dico "SIGLE -> Definition" ---
    Set dicoImport = CreateObject("Scripting.Dictionary")

    numFichier = FreeFile

    On Error GoTo GestionErreurImport
    Open cheminFichier For Input As #numFichier
    Do While Not EOF(numFichier)
        Line Input #numFichier, ligneFichier
        ligneFichier = Trim(ligneFichier)
        If Len(ligneFichier) > 0 Then
            ' [v133] séparateur : | puis tab, ; , : — le premier dont la partie gauche
            ' est un vrai sigle (sans espace, 15 car. max). Sinon ligne ignorée.
            posSep = 0
            Dim sepsImp As Variant, sp As Variant
            sepsImp = Array(SEPARATEUR_GLOSSAIRE_PERSO, vbTab, ";", ":")
            For Each sp In sepsImp
                posSep = InStr(ligneFichier, CStr(sp))
                If posSep > 1 Then
                    sigleLu = Trim(UCase(Left(ligneFichier, posSep - 1)))
                    If Len(sigleLu) > 0 And Len(sigleLu) <= 15 And InStr(sigleLu, " ") = 0 Then Exit For
                End If
                posSep = 0
            Next sp
            If posSep > 0 Then
                sigleLu = Trim(UCase(Left(ligneFichier, posSep - 1)))
                defLue = Trim(Mid(ligneFichier, posSep + 1))
                If Len(sigleLu) > 0 And Len(defLue) > 0 Then
                    If Not dicoImport.Exists(sigleLu) Then
                        dicoImport.Add sigleLu, defLue
                    End If
                End If
            End If
        End If

    On Error GoTo 0

    End If

            End If

        If tbl Is Nothing Then
            MsgBox "Création du tableau impossible.", vbExclamation
            Exit Sub
        End If
        Set tbl = TrouverTableauSiglesPourAjout(posSigleT, posSignifT)
        If tbl Is Nothing Then
            MsgBox "Tableau créé mais non reconnu : import impossible.", vbExclamation
            Exit Sub
        End If

    ' [v131] Option : n'importer que les sigles réellement présents dans le document
    Dim repFiltre As VbMsgBoxResult
    Dim dicoUtilises As Object
    Dim cleF As Variant
    Dim rngF As Range
    Dim nbRetires As Long
    repFiltre = MsgBox(dicoImport.Count & " sigle(s) dans le fichier." & vbCrLf & vbCrLf & _
        "N'importer QUE ceux présents dans le document (rouges, non traités, ou cités dans le tableau) ?" & vbCrLf & _
        "Oui = seulement les sigles utilisés" & vbCrLf & "Non = tous les sigles du fichier", _
        vbYesNoCancel + vbQuestion, "Import du glossaire")
    If repFiltre = vbCancel Then Exit Sub

        For Each cleF In dicoImport.Keys
            Set rngF = ActiveDocument.Content
            With rngF.Find
                .ClearFormatting

                .MatchCase = True
                .MatchWholeWord = True
                .MatchWildcards = False
                .Forward = True
                .Wrap = wdFindStop
            End With
            If rngF.Find.Execute Then dicoUtilises.Add CStr(cleF), dicoImport(cleF)
        Next cleF

        nbRetires = dicoImport.Count - dicoUtilises.Count
        Set dicoImport = dicoUtilises

            MsgBox "Aucun sigle du fichier n'est présent dans le document.", vbInformation
            Exit Sub
        End If
    End If

    Set dicoExistant = CreateObject("Scripting.Dictionary")
    On Error Resume Next
    For rLigneImp = 2 To tbl.Rows.Count
        texteExistantImp = Trim(UCase(NettoyerTexteCellule(tbl.Cell(rLigneImp, posSigleT).Range.Text)))

    On Error GoTo 0

    Application.ScreenUpdating = False

                nbImportes = nbImportes + 1
                dicoAjoutesImp.Add UCase(CStr(cleImport)), True
                dicoExistant.Add CStr(cleImport), True
                ' [v102] Le sigle est désormais connu : les occurrences
                ' encore en rouge deviennent des liens avec infobulle.

                nbRougesConverties = nbRougesConverties + AppliquerDefinitionAuxOccurrencesRouges(CStr(cleImport), CStr(dicoImport(cleImport)))
            Else
                nbIgnores = nbIgnores + 1
            End If
        End If
    Next cleImport

    ' [v126] Traite les définitions des lignes importées (sigles cités)
    Application.StatusBar = "Traitement des définitions importées..."
    nbLiensDefs = TraiterDefinitionsAjoutees(dicoAjoutesImp)

    On Error Resume Next
    Application.UndoRecord.EndCustomRecord
    Application.StatusBar = False
    ActiveDocument.TrackRevisions = ancienSuiviModifImp
    On Error GoTo 0

' [v98] Exporte le tableau de sigles du document courant vers un
' fichier glossaire personnel externe (.txt, format
' "SIGLE|Definition"). Si le fichier choisi existe déjà, FUSION
' automatique et silencieuse : les entrées déjà présentes dans le
' fichier mais absentes du tableau sont conservées, celles du
' tableau sont ajoutées/mises à jour. Le tableau reste AUTORITAIRE
' (cohérent avec la règle v93) : en cas de sigle présent des deux
' côtés avec une définition différente, celle du tableau l'emporte
' systématiquement.
Sub ExporterGlossairePersonnel()

    ' --- Variables ---
    Dim posSigleE As Long
    Dim posSignifE As Long
    Dim tblE As Table
    Dim cheminExport As String
    Dim dicoFusion As Object
    Dim numFichierE As Integer
    Dim ligneE As String
    Dim posSepE As Long
    Dim sigleLuE As String
    Dim defLueE As String
    Dim rLigneE As Long
    Dim sigleTableE As String
    Dim defTableE As String
    Dim numFichierW As Integer
    Dim cleFusion As Variant

    Set tblE = TrouverTableauSiglesPourAjout(posSigleE, posSignifE)

    If tblE Is Nothing Then
        MsgBox "Aucun tableau de sigles reconnu dans ce document : export impossible.", vbExclamation
        Exit Sub
    End If

    cheminExport = ""
    With Application.FileDialog(msoFileDialogSaveAs)
        .Title = "Choisir où enregistrer le glossaire personnel"
        .InitialFileName = "glossaire_perso.txt"
        If .Show = -1 Then cheminExport = .SelectedItems(1)
    End With

    If Len(cheminExport) = 0 Then Exit Sub ' annulé par l'utilisateur

    ' --- Chargement du contenu existant du fichier, s'il existe (fusion) ---
    Set dicoFusion = CreateObject("Scripting.Dictionary")

    If Len(Dir(cheminExport)) > 0 Then
        numFichierE = FreeFile
        On Error Resume Next
        Open cheminExport For Input As #numFichierE
        Do While Not EOF(numFichierE)
            Line Input #numFichierE, ligneE
            ligneE = Trim(ligneE)
            If Len(ligneE) > 0 Then
                posSepE = InStr(ligneE, SEPARATEUR_GLOSSAIRE_PERSO)
                If posSepE > 0 Then
                    sigleLuE = Trim(UCase(Left(ligneE, posSepE - 1)))
                    defLueE = Trim(Mid(ligneE, posSepE + 1))
                    If Len(sigleLuE) > 0 And Not dicoFusion.Exists(sigleLuE) Then
                        dicoFusion.Add sigleLuE, defLueE
                    End If
                End If
            End If
        Loop
        Close #numFichierE
        On Error GoTo 0
    End If

    ' --- Entrées du tableau : AUTORITAIRES, écrasent celles du fichier ---
    On Error Resume Next
    For rLigneE = 2 To tblE.Rows.Count
        sigleTableE = Trim(UCase(NettoyerTexteCellule(tblE.Cell(rLigneE, posSigleE).Range.Text)))
        defTableE = Trim(NettoyerTexteCellule(tblE.Cell(rLigneE, posSignifE).Range.Text))
        If Len(sigleTableE) > 0 Then
            If dicoFusion.Exists(sigleTableE) Then
                dicoFusion(sigleTableE) = defTableE
            Else
                dicoFusion.Add sigleTableE, defTableE
            End If
        End If
    Next rLigneE
    On Error GoTo 0

    ' --- Écriture du fichier final ---
    numFichierW = FreeFile
    On Error GoTo GestionErreurExport
    Open cheminExport For Output As #numFichierW
    For Each cleFusion In dicoFusion.Keys
        Print #numFichierW, cleFusion & SEPARATEUR_GLOSSAIRE_PERSO & dicoFusion(cleFusion)
    Next cleFusion
    Close #numFichierW
    On Error GoTo 0

    MsgBox dicoFusion.Count & " sigle(s) au total dans le glossaire personnel." & vbCrLf & _
           "Fichier enregistré : " & cheminExport, vbInformation, "Export terminé"

    Exit Sub

GestionErreurExport:
    On Error Resume Next
    Close #numFichierW
    On Error GoTo 0
    MsgBox "Erreur lors de l'écriture du fichier glossaire : " & Err.Description, vbCritical
End Sub

' [v81] FACTORISATION : matérialise Start/End/Texte de tous les
' mots du document dans des tableaux en mémoire (accès rapide
' sans appel COM répété). Remplace un bloc dupliqué dans
' TraiterDocumentUnePasse et SurlignerRougeSeul.
Private Function EstSeparateurInsecable(c As String) As Boolean
    ' [v117] Espaces que Word ne sépare pas toujours des mots voisins :
    ' insécable (160), fine (8201), fine insécable (8239), demi/cadratin, tab.
    Dim n As Long
    n = AscW(c)
    EstSeparateurInsecable = (n = 160 Or n = 8239 Or n = 8201 Or n = 8194 Or n = 8195 Or n = 8202 Or n = 9 Or n = 11)
End Function

' [v117] Ajoute un "mot" aux tableaux (croissance automatique), après
' retrait de l'élision (l'APX) et de la ponctuation initiale ("(ONU").
Private Sub PousserMot(ByRef wStart() As Long, ByRef wFin() As Long, ByRef wTexte() As String, ByRef nb As Long, ByRef cap As Long, ByVal deb As Long, ByVal fin As Long, ByVal txt As String)
    Dim posApo As Long
    Dim posAln As Long
    Dim iAln As Long

    ' Élision : "l'APX" / "d’APX" -> "APX"
    posApo = InStrRev(txt, "'")
    If InStrRev(txt, ChrW(8217)) > posApo Then posApo = InStrRev(txt, ChrW(8217))
    If posApo > 0 And posApo < Len(txt) Then
        deb = deb + posApo
        txt = Mid(txt, posApo + 1)
    End If

    ' Ponctuation initiale (uniquement si le mot contient de l'alphanumérique)
    posAln = 0
    For iAln = 1 To Len(txt)
        If EstCaractereMot(Mid(txt, iAln, 1)) Then
            posAln = iAln
            Exit For
        End If
    Next iAln
    If posAln > 1 Then
        deb = deb + posAln - 1
        txt = Mid(txt, posAln)
    End If

    ' [v118] Fin de cellule / fin de paragraphe : Word colle la marque
    ' (Chr(13)+Chr(7), 1 seule position) au dernier mot. On la retire
    ' du texte ET de la plage, pour que la plage reste DANS la cellule.
    If Len(txt) >= 2 Then
        If Right(txt, 2) = Chr(13) & Chr(7) Then
            txt = Left(txt, Len(txt) - 2)
            fin = fin - 1
        End If
    End If
    If Len(txt) >= 1 Then
        If Right(txt, 1) = Chr(7) Or Right(txt, 1) = Chr(13) Then
            txt = Left(txt, Len(txt) - 1)
            fin = fin - 1
        End If
    End If
    If Len(txt) = 0 Then Exit Sub

    nb = nb + 1
    If nb > cap Then
        cap = cap + 5000
        ReDim Preserve wStart(1 To cap)
        ReDim Preserve wFin(1 To cap)
        ReDim Preserve wTexte(1 To cap)
    End If
    wStart(nb) = deb
    wFin(nb) = fin
    wTexte(nb) = txt
End Sub

' [v81] FACTORISATION : matérialise Start/End/Texte de tous les
' mots du document dans des tableaux en mémoire.
' [v117] Un "mot" Word contenant une espace insécable / fine
' (« du TESE », « ONU ») est découpé en plusieurs mots :
' sinon "du TESE" n'était jamais reconnu comme contenant le sigle.
Private Sub MaterialiserMots(ByRef wStart() As Long, ByRef wFin() As Long, ByRef wTexte() As String, ByRef totalWords As Long, Optional rngSource As Range = Nothing)
    Dim wrdTmp As Range
    Dim src As Range
    Dim nbMotsSrc As Long
    Dim cap As Long
    Dim nb As Long
    Dim nbLus As Long
    Dim txtW As String
    Dim lw As Long
    Dim i As Long
    Dim debP As Long
    Dim baseW As Long
    Dim finP As Long
    Dim coupe As Boolean

    If rngSource Is Nothing Then
        Set src = ActiveDocument.Content
    Else
        Set src = rngSource
    End If
    nbMotsSrc = src.Words.Count
    cap = nbMotsSrc + 1000
    ReDim wStart(1 To cap)
    ReDim wFin(1 To cap)
    ReDim wTexte(1 To cap)
    nb = 0
    nbLus = 0

    For Each wrdTmp In src.Words
        nbLus = nbLus + 1
        txtW = wrdTmp.Text
        lw = Len(txtW)
        baseW = wrdTmp.Start
        debP = 1
        For i = 1 To lw + 1
            If i > lw Then
                coupe = True
            Else
                coupe = EstSeparateurInsecable(Mid(txtW, i, 1))
            End If
            If coupe Then
                If i > debP Then
                    If i > lw And debP = 1 Then
                        finP = wrdTmp.End          ' mot entier : comportement d'origine
                    ElseIf i > lw Then
                        finP = wrdTmp.End
                    Else
                        finP = baseW + i - 1
                    End If
                    PousserMot wStart, wFin, wTexte, nb, cap, baseW + debP - 1, finP, Mid(txtW, debP, i - debP)
                End If
                debP = i + 1
            End If
        Next i
        If nbLus Mod 3000 = 0 Then
            Application.StatusBar = "Lecture du document : " & nbLus & "/" & nbMotsSrc
            DoEvents
        End If
    Next wrdTmp

    totalWords = nb
End Sub

' [v73] Enregistre (ou met à jour) une définition manuelle pour

Private Sub AjouterDefinitionFichier(sigle As String, definition As String)

    sigleNorm = Trim(UCase(sigle))

    If Len(definition) > 80 Then definition = Left(definition, 80)

    Else
        dicoDef.Add sigleNorm, definition
    End If

    Next cleDef

    On Error Resume Next

    On Error GoTo 0
    ActiveDocument.Variables.Add NOM_VARIABLE_DEFINITIONS, nouveauContenu
End Sub

    On Error GoTo 0
    If Len(nouveauContenu) > 0 Then
        ActiveDocument.Variables.Add NOM_VARIABLE_DEFINITIONS, nouveauContenu
    End If
End Sub

' [v130] SÉLECTION MULTIPLE (Ctrl+sélection) : Selection.Range ne renvoie
' qu'une zone. Astuce : Selection.Font s'applique à TOUTES les zones ;
' on pose une couleur de police-témoin, on retrouve les zones par Find
' (format), on lit leur texte puis on remet la couleur automatique.
Private Function RecolterSelectionMultiple() As String
    Dim rng As Range
    Dim marque As Long
    Dim dS() As Long, dF() As Long
    Dim n As Long, k As Long
    Dim res As String

    If Selection.Start = Selection.End Then Exit Function
    Dim tDeb As Single
    tDeb = Timer
    marque = RGB(1, 2, 3)
    ReDim dS(0 To 500)
    ReDim dF(0 To 500)

    On Error GoTo Secours
    Application.ScreenUpdating = False
    Selection.Font.Color = marque

    Set rng = ActiveDocument.Content
    With rng.Find
        .ClearFormatting
        .Text = ""
        .Font.Color = marque
        .Format = True
        .Forward = True
        .Wrap = wdFindStop
        .MatchWildcards = False
    End With
    Do While rng.Find.Execute
        If n > UBound(dS) Then
            ReDim Preserve dS(0 To n + 500)
            ReDim Preserve dF(0 To n + 500)
        End If
        dS(n) = rng.Start
        dF(n) = rng.End
        res = res & " " & rng.Text
        n = n + 1
        rng.Collapse wdCollapseEnd
        If n > 5000 Then Exit Do
        If Timer - tDeb > 4 Then Exit Do
    Loop

    For k = 0 To n - 1
        ActiveDocument.Range(dS(k), dF(k)).Font.Color = wdColorAutomatic
    Next k
    Application.ScreenUpdating = True
    ' [v142] toujours ajouter le texte de la sélection principale (repli si Find ne trouve rien)
    RecolterSelectionMultiple = res & " " & Selection.Text
    Exit Function

Secours:
    On Error Resume Next
    Application.ScreenUpdating = True
    RecolterSelectionMultiple = Selection.Text
End Function

' [v128] Accepte UNE ou PLUSIEURS sigles dans la sélection (ex. un
' paragraphe, une liste) : chaque sigle détecté est ajouté aux exceptions.
Sub AjouterExceptionDepuisSelection()
    Dim txt As String
    Dim lst As Object
    Dim i As Long
    Dim c As String
    Dim jeton As String
    Dim cle As Variant
    Dim apercu As String
    Dim rep As VbMsgBoxResult

    txt = RecolterSelectionMultiple()
    If Len(Trim(txt)) = 0 Then
        MsgBox "Sélectionnez d'abord un ou plusieurs sigles dans le document.", vbExclamation
        Exit Sub
    End If

    Set lst = CreateObject("Scripting.Dictionary")
    lst.CompareMode = vbTextCompare
    jeton = ""
    For i = 1 To Len(txt) + 1
        If i <= Len(txt) Then c = Mid(txt, i, 1) Else c = " "
        If EstCaractereMot(c) Then
            jeton = jeton & c
        Else
            If Len(jeton) > 0 Then
                If EstUnSigle(jeton) Then
                    If Not lst.Exists(jeton) Then lst.Add jeton, True
                End If
                jeton = ""
            End If
        End If
    Next i

    If lst.Count = 0 Then
        Dim mot1 As String
        mot1 = RetirerPonctuationFinale(Trim(txt))
        If Len(mot1) = 0 Or Len(mot1) > 30 Then
            MsgBox "Aucun sigle trouvé dans la sélection.", vbExclamation
            Exit Sub
        End If
        rep = MsgBox("Le texte sélectionné (""" & mot1 & """) ne ressemble pas à un sigle (2-15 lettres/chiffres majuscules)." & vbCrLf & "L'ajouter quand même à la liste des exceptions ?", vbYesNo + vbQuestion)
        If rep = vbNo Then Exit Sub
        lst.Add mot1, True
    ElseIf lst.Count > 1 Then
        For Each cle In lst.Keys
            apercu = apercu & cle & "  "
        Next cle
        rep = MsgBox(lst.Count & " sigles trouvés dans la sélection :" & vbCrLf & vbCrLf & apercu & vbCrLf & vbCrLf & "Tous les ajouter aux exceptions ?", vbYesNo + vbQuestion, "Exceptions multiples")
        If rep = vbNo Then Exit Sub
    End If

    ' [v141] Bascule : si TOUS les sigles sélectionnés sont déjà des exceptions,
    ' proposer de les RETIRER des exceptions.
    Dim dExc As Object
    Dim tousDeja As Boolean
    Set dExc = ChargerExceptions()
    tousDeja = (lst.Count > 0)
    For Each cle In lst.Keys
        If Not dExc.Exists(UCase(CStr(cle))) Then tousDeja = False: Exit For
    Next cle
    If tousDeja Then
        apercu = ""
        For Each cle In lst.Keys
            apercu = apercu & cle & "  "
        Next cle
        rep = MsgBox("Déjà dans les exceptions :" & vbCrLf & vbCrLf & apercu & vbCrLf & vbCrLf & _
                     "Les RETIRER des exceptions ?" & vbCrLf & "(ils seront de nouveau détectés au prochain Traiter)", _
                     vbYesNo + vbQuestion, "Exceptions")
        If rep = vbYes Then
            For Each cle In lst.Keys
                RetirerExceptionFichier CStr(cle)
            Next cle
            MsgBox lst.Count & " sigle(s) retiré(s) des exceptions. Relancez Traiter pour les détecter." & vbCrLf & _
                   "Pensez à enregistrer (Ctrl+S).", vbInformation
        End If
        Exit Sub
    End If

    AppliquerListeExceptions lst, Selection.Range
End Sub

' [v141] Retire un mot de la liste des exceptions du document.
Private Sub RetirerExceptionFichier(mot As String)
    Dim dicoE As Object
    Dim cle As Variant
    Dim contenu As String
    Set dicoE = ChargerExceptions()
    If dicoE.Exists(Trim(UCase(mot))) Then dicoE.Remove Trim(UCase(mot))
    For Each cle In dicoE.Keys
        If Len(contenu) > 0 Then contenu = contenu & SEPARATEUR_EXCEPTIONS
        contenu = contenu & cle
    Next cle
    On Error Resume Next
    ActiveDocument.Variables(NOM_VARIABLE_EXCEPTIONS).Delete
    On Error GoTo 0
    If Len(contenu) > 0 Then ActiveDocument.Variables.Add NOM_VARIABLE_EXCEPTIONS, contenu
End Sub

' [v129] Ajoute une liste de sigles (Dictionary) aux exceptions et nettoie le document.
Private Sub AppliquerListeExceptions(lst As Object, rngS As Range)
    Dim ancienSuiviM As Boolean
    Dim cle As Variant
    Dim nbCorr As Long, nbAj As Long
    For Each cle In lst.Keys
        AjouterExceptionFichier CStr(cle)
        nbAj = nbAj + 1
    Next cle

    On Error Resume Next
    ancienSuiviM = ActiveDocument.TrackRevisions
    ActiveDocument.TrackRevisions = False
    Application.UndoRecord.StartCustomRecord "Ajout exceptions"
    On Error GoTo 0
    Application.ScreenUpdating = False

    ' la sélection est nettoyée une fois (rouge), puis chaque sigle
    For Each cle In lst.Keys
        nbCorr = nbCorr + ExceptionPourMot(CStr(cle), rngS)
    Next cle

    On Error Resume Next
    Application.UndoRecord.EndCustomRecord
    ActiveDocument.TrackRevisions = ancienSuiviM
    On Error GoTo 0
    Application.ScreenUpdating = True

    MsgBox nbAj & " sigle(s) ajouté(s) aux exceptions de CE document." & vbCrLf & _
           nbCorr & " occurrence(s) nettoyée(s) (lien/rouge retiré)." & vbCrLf & vbCrLf & _
           "Ignorés lors des prochains traitements. Pensez à enregistrer (Ctrl+S).", vbInformation

End Sub

Private Function ExceptionPourMot(mot As String, rngSel As Range) As Long

    Dim rngRecherche As Range
    Dim nbCorrige As Long
    nbCorrige = 0

    ' [v108] Nettoie d'abord la SÉLECTION elle-même (cas d'un sigle
    ' dans un tableau, où la recherche Find peut le manquer).
    On Error Resume Next
    If rngSel.Words.Count <= 2 Then
        If RetirerRougeSurPlage(rngSel) Then nbCorrige = nbCorrige + 1
    End If
    On Error GoTo 0

    ' [v71] CORRECTIF BOUCLE INFINIE : la version précédente
    Dim posDebutsExcept() As Long
    Dim posFinsExcept() As Long
    Dim nbTrouvesExcept As Long
    ReDim posDebutsExcept(0 To 200)
    ReDim posFinsExcept(0 To 200)
    nbTrouvesExcept = 0

    Set rngRecherche = ActiveDocument.Content

    Do While rngRecherche.Find.Execute(FindText:=mot, MatchWholeWord:=True, MatchCase:=False, Forward:=True, Wrap:=wdFindStop, MatchWildcards:=False)

        If nbTrouvesExcept > UBound(posDebutsExcept) Then
            ReDim Preserve posDebutsExcept(0 To nbTrouvesExcept + 200)
            ReDim Preserve posFinsExcept(0 To nbTrouvesExcept + 200)
        End If
        posDebutsExcept(nbTrouvesExcept) = rngRecherche.Start
        posFinsExcept(nbTrouvesExcept) = rngRecherche.End
        nbTrouvesExcept = nbTrouvesExcept + 1

        rngRecherche.Collapse wdCollapseEnd

        ' Sécurité anti-boucle infinie
        If nbTrouvesExcept > 5000 Then Exit Do

    Loop

    ' --- Nettoyage en ordre inverse (de la fin vers le début) ---
    Dim kExcept As Long
    For kExcept = nbTrouvesExcept - 1 To 0 Step -1

        Dim rngCible As Range
        Set rngCible = ActiveDocument.Range(posDebutsExcept(kExcept), posFinsExcept(kExcept))

        Dim futRetouche As Boolean
        futRetouche = False

        ' Retire le surlignage rouge s'il y en a un
        ' [v108] Caractère par caractère (plage mixte ou en tableau)
        If RetirerRougeSurPlage(rngCible) Then futRetouche = True

        ' Retire le lien s'il y en a un (via Unlink sur le champ,
        ' méthode la plus fiable établie précédemment)
        If rngCible.Hyperlinks.Count > 0 Then
            Dim fldExcept As Field
            On Error Resume Next
            For Each fldExcept In rngCible.Fields
                If fldExcept.Type = wdFieldHyperlink Then
                    fldExcept.Result.Font.Shading.Texture = wdTextureNone
                    fldExcept.Unlink
                End If
            Next fldExcept
            rngCible.Font.Shading.Texture = wdTextureNone
            On Error GoTo 0
            futRetouche = True
        End If

        If futRetouche Then nbCorrige = nbCorrige + 1

    Next kExcept

    ExceptionPourMot = nbCorrige
End Function

' [v129] Sélection MULTIPLE par accumulation : chaque appel ajoute le
' mot sélectionné à une pile en attente (variable du document).
Sub MarquerExceptionPile()
    Dim mot As String
    Dim pile As String
    mot = RetirerPonctuationFinale(Trim(Selection.Text))
    If Len(mot) = 0 Or Len(mot) > 30 Then
        MsgBox "Sélectionnez un sigle à marquer.", vbExclamation
        Exit Sub
    End If
    pile = LirePileExceptions()
    If InStr(1, "|" & pile & "|", "|" & mot & "|", vbTextCompare) = 0 Then
        If Len(pile) > 0 Then pile = pile & "|"
        pile = pile & mot
        ActiveDocument.Variables("SiglesPile").Value = pile
    End If
    Application.StatusBar = "Exceptions en attente (" & (UBound(Split(pile, "|")) + 1) & ") : " & Replace(pile, "|", ", ")
End Sub

' Valide la pile : tous les sigles marqués deviennent des exceptions.
Sub ValiderPileExceptions()
    Dim pile As String
    Dim p As Variant
    Dim lst As Object
    Dim rep As VbMsgBoxResult
    pile = LirePileExceptions()
    If Len(pile) = 0 Then
        MsgBox "Aucun sigle en attente. Utilisez d'abord ""Marquer"".", vbExclamation
        Exit Sub
    End If
    Set lst = CreateObject("Scripting.Dictionary")
    lst.CompareMode = vbTextCompare
    For Each p In Split(pile, "|")
        If Len(p) > 0 Then If Not lst.Exists(CStr(p)) Then lst.Add CStr(p), True
    Next p
    rep = MsgBox(lst.Count & " sigle(s) en attente :" & vbCrLf & vbCrLf & Replace(pile, "|", "  ") & vbCrLf & vbCrLf & "Les ajouter aux exceptions ?" & vbCrLf & "(Non = vider la pile)", vbYesNoCancel + vbQuestion, "Exceptions multiples")
    If rep = vbCancel Then Exit Sub
    On Error Resume Next
    ActiveDocument.Variables("SiglesPile").Delete
    On Error GoTo 0
    Application.StatusBar = False
    If rep = vbNo Then Exit Sub
    AppliquerListeExceptions lst, Selection.Range
End Sub

' =========================================================
' Construit le dictionnaire à partir des tableaux du document
' =========================================================


Private Sub ChargerDicoDepuisTableaux(dico As Object, ByRef tblDebuts() As Long, ByRef tblFins() As Long, ByRef tblColSigle() As Long, ByRef nbTables As Long, Optional motCleSigleUtilisateur As String = "", Optional motCleSignifUtilisateur As String = "")

    Dim tbl As Table
    Dim cel As Cell
    Dim posSigle As Long
    Dim posSignif As Long
    Dim posActuelle As Long
    Dim texteEntete As String
    Dim sigle As String
    Dim signif As String
    Dim premiereLigne As Boolean
    Dim numLigne As Long
    Dim numTbl As Long
    Dim rIdx As Long
    Dim rIdxPrecedent As Long

    numTbl = 0
    nbTables = 0
    ReDim tblDebuts(0 To 50)
    ReDim tblFins(0 To 50)
    ReDim tblColSigle(0 To 50)

    For Each tbl In ActiveDocument.Tables

        numTbl = numTbl + 1
        posSigle = 0
        posSignif = 0
        premiereLigne = True
        numLigne = 0
        rIdxPrecedent = -1
        posActuelle = 0
        sigle = ""
        signif = ""

        ' =====================================================
        ' [v38] "For Each lig In tbl.Rows" pouvait lever une
        ' =====================================================
        On Error GoTo TableIgnoree

        For Each cel In tbl.Range.Cells

            rIdx = cel.RowIndex

            If rIdx <> rIdxPrecedent Then
                ' --- Nouvelle ligne : finalise la ligne précédente ---
                If rIdxPrecedent <> -1 Then
                    If premiereLigne Then
                        premiereLigne = False
                    Else
                        sigle = Trim(sigle)
                        signif = Trim(signif)
                        If Len(signif) > 80 Then signif = Left(signif, 80)

                        If posSigle > 0 And posSignif > 0 Then
                            If Len(sigle) >= 2 And Len(sigle) <= 15 And Len(signif) >= 2 Then
                                If Not dico.Exists(sigle) Then
                                    dico.Add sigle, signif
                                End If
                            End If
                        End If
                    End If

                    numLigne = numLigne + 1
                    Application.StatusBar = "Tableau " & numTbl & " - ligne " & numLigne & " (" & dico.Count & " sigles trouvés)"
                    DoEvents
                End If

                posActuelle = 0
                sigle = ""
                signif = ""
                rIdxPrecedent = rIdx
            End If

            posActuelle = posActuelle + 1

            ' [v64] Utilise cel.ColumnIndex (la VRAIE colonne Word)
            Dim colReelle As Long
            colReelle = posActuelle
            On Error Resume Next
            colReelle = cel.ColumnIndex
            On Error GoTo 0

            If premiereLigne Then
                texteEntete = ""
                texteEntete = NettoyerTexteCellule(cel.Range.Text)

                ' [v40] Liste de mots-clés élargie pour couvrir plus
                If posSigle = 0 Then
                    If ContientMotCle(texteEntete, Array("sigle", "acronyme", "abreviation", "abréviation", "code", "trigramme", "raccourci", "terme", "mot")) Then
                        posSigle = colReelle
                    ElseIf Len(motCleSigleUtilisateur) > 0 Then
                        If StrComp(Trim(texteEntete), Trim(motCleSigleUtilisateur), vbTextCompare) = 0 Then
                            posSigle = colReelle
                        End If
                    End If
                End If
                If posSignif = 0 Then
                    If ContientMotCle(texteEntete, Array("signification", "definition", "définition", "libelle", "libellé", "description", "sens", "explication", "traduction", "developpe", "développé", "detail", "détail", "texte")) Then
                        posSignif = colReelle
                    ElseIf Len(motCleSignifUtilisateur) > 0 Then
                        If StrComp(Trim(texteEntete), Trim(motCleSignifUtilisateur), vbTextCompare) = 0 Then
                            posSignif = colReelle
                        End If
                    End If
                End If

            Else
                If colReelle = posSigle Then
                    sigle = NettoyerTexteCellule(cel.Range.Text)
                End If
                If colReelle = posSignif Then
                    signif = NettoyerTexteCellule(cel.Range.Text)
                End If
            End If

        Next cel

        ' [v59] Mémorise Start/End du tableau + sa colonne "SIGLE"
        If posSigle > 0 Then
            If nbTables > UBound(tblDebuts) Then
                ReDim Preserve tblDebuts(0 To nbTables + 50)
                ReDim Preserve tblFins(0 To nbTables + 50)
                ReDim Preserve tblColSigle(0 To nbTables + 50)
            End If
            tblDebuts(nbTables) = tbl.Range.Start
            tblFins(nbTables) = tbl.Range.End
            tblColSigle(nbTables) = posSigle
            nbTables = nbTables + 1
        End If

        ' --- Finalise la toute dernière ligne du tableau ---
        If Not premiereLigne Then
            sigle = Trim(sigle)
            signif = Trim(signif)
            If Len(signif) > 80 Then signif = Left(signif, 80)

            If posSigle > 0 And posSignif > 0 Then
                If Len(sigle) >= 2 And Len(sigle) <= 15 And Len(signif) >= 2 Then
                    If Not dico.Exists(sigle) Then
                        dico.Add sigle, signif
                    End If
                End If
            End If
        End If

        On Error GoTo 0
        GoTo TableSuivante

TableIgnoree:
        Debug.Print "Tableau " & numTbl & " IGNORÉ (erreur " & Err.Number & " : " & Err.Description & ") — probablement des cellules fusionnées verticalement."
        Err.Clear
        On Error GoTo 0

TableSuivante:
    Next tbl

End Sub
' =========================================================
' Détection complémentaire par parenthèses (sigles absents
' =========================================================


Private Sub DetecterSiglesManuel(dico As Object)

    Dim texte As String
    Dim posOuvrante As Long
    Dim posFermante As Long
    Dim contenu As String
    Dim avant As String
    Dim sigle As String
    Dim signif As String

    texte = ActiveDocument.Content.Text
    posOuvrante = InStr(1, texte, "(")

    Do While posOuvrante > 0

        posFermante = InStr(posOuvrante, texte, ")")
        If posFermante = 0 Then Exit Do

        contenu = Mid(texte, posOuvrante + 1, posFermante - posOuvrante - 1)

        If Not PositionDansTableau(posOuvrante) Then

            If EstUnSigle(contenu) Then
                sigle = contenu
                If Not dico.Exists(sigle) Then
                    avant = Left(texte, posOuvrante - 1)
                    signif = ExtraireGroupeNominal(avant)
                    If Len(signif) > 0 Then
                        dico.Add sigle, signif
                    End If
                End If
            Else
                avant = Left(texte, posOuvrante - 1)
                sigle = ExtraireDernierMot(avant)
                If EstUnSigle(sigle) And Len(contenu) >= 3 Then
                    If Not dico.Exists(sigle) Then
                        Dim sc As String
                        sc = Trim(contenu)
                        If Len(sc) > 80 Then sc = Left(sc, 80)
                        dico.Add sigle, sc
                    End If
                End If
            End If

        End If

        posOuvrante = InStr(posFermante + 1, texte, "(")
    Loop

End Sub

Sub ValiderTraitementSigles()

' =========================================================
' Supprime TOUS les hyperliens du document en retirant le
' =========================================================


Sub SupprimerTousLesLiensGeneres()

    Dim fld As Field
    Dim compteurTotal As Long
    Dim k As Long
    Dim rep As VbMsgBoxResult
    Dim nbAvant As Long
    Dim nbApres As Long
    Dim passage As Long
    Dim journalErreurs As String

    rep = MsgBox("Supprimer tous les liens générés par la macro et réinitialiser la mise en forme (gris + rouge) ?", vbYesNo + vbQuestion, "Confirmation")
    If rep = vbNo Then Exit Sub

    ' [v109] Question séparée : supprimer aussi les exceptions ?
    Dim supprimerExceptions As Boolean
    supprimerExceptions = False
    If ChargerExceptions().Count > 0 Then
        supprimerExceptions = (MsgBox("Supprimer aussi les " & ChargerExceptions().Count & " exception(s) enregistrée(s) dans ce document ?", vbYesNo + vbQuestion, "Exceptions") = vbYes)
    End If

    Application.ScreenUpdating = False
    compteurTotal = 0
    passage = 0
    journalErreurs = ""

    ' [v79] Désactive temporairement le suivi des modifications
    ' (même raison que v78 : évite une bulle par champ retiré).
    Dim ancienSuiviModifSup As Boolean
    On Error Resume Next
    ancienSuiviModifSup = ActiveDocument.TrackRevisions
    ActiveDocument.TrackRevisions = False
    On Error GoTo 0

    ' [v41] Regroupe toutes les suppressions dans un seul bloc
    On Error Resume Next
    Application.UndoRecord.StartCustomRecord "Suppression des liens generes"
    On Error GoTo 0

    ' --- Nettoyage de la mise en forme AVANT suppression des champs ---
    On Error Resume Next
    ActiveDocument.Range.HighlightColorIndex = wdNoHighlight
    On Error GoTo 0

    ' --- Boucle de suppression via Fields, répétée jusqu'à stabilisation ---
    Do

        passage = passage + 1
        nbAvant = CompterLiensGeneres()
        If nbAvant = 0 Then Exit Do

        For k = ActiveDocument.Fields.Count To 1 Step -1

            On Error Resume Next
            Err.Clear
            Set fld = ActiveDocument.Fields(k)

            If fld.Type = wdFieldHyperlink And Not EstLienInterne(fld) Then

                ' Retire la mise en forme avant de délier le champ
                fld.Result.Font.Shading.Texture = wdTextureNone
                fld.Result.Font.ColorIndex = wdAuto
                fld.Result.Font.Underline = wdUnderlineNone

                fld.Unlink

                If Err.Number <> 0 Then
                    journalErreurs = journalErreurs & "Champ #" & k & " - Erreur " & Err.Number & " : " & Err.Description & vbCrLf
                    Debug.Print "Echec Unlink champ #" & k & " - Erreur " & Err.Number & " : " & Err.Description
                End If

            End If
            On Error GoTo 0

            If k Mod 50 = 0 Then
                Application.StatusBar = "Nettoyage (passage " & passage & ") : " & k & " champ(s) restant(s) à examiner"
                DoEvents
            End If

        Next k

        nbApres = CompterLiensGeneres()
        compteurTotal = compteurTotal + (nbAvant - nbApres)

        If nbApres = 0 Then Exit Do
        If nbApres = nbAvant Then Exit Do ' stagnation : abandon
        If passage >= 10 Then Exit Do

    Loop

    ActiveDocument.Range.HighlightColorIndex = wdNoHighlight

    Application.StatusBar = False
    Application.ScreenUpdating = True

    If CompterLiensGeneres() > 0 Then

        Dim rapportEchec As String
        Dim m As Long
        Dim adrM As String
        rapportEchec = "Liens résistants (" & CompterLiensGeneres() & " au total) :" & vbCrLf & vbCrLf
        For m = 1 To ActiveDocument.Hyperlinks.Count
            adrM = ""
            On Error Resume Next
            adrM = ActiveDocument.Hyperlinks(m).Address
            On Error GoTo 0
            rapportEchec = rapportEchec & m & ". Texte=[" & ActiveDocument.Hyperlinks(m).Range.Text & "] Adresse=[" & adrM & "]" & vbCrLf
        Next m

        ' [v32] Rapport copiable désactivé (usage production).

        MsgBox compteurTotal & " lien(s) supprimé(s) via Unlink, mais " & ActiveDocument.Hyperlinks.Count & " subsiste(nt) encore et n'ont pas pu être retirés.", vbExclamation
    Else
        MsgBox compteurTotal & " lien(s) supprimé(s) au total (via Unlink). Document réinitialisé (prêt pour un nouveau test).", vbInformation, "Nettoyage terminé"
    End If

    ' [v105] Supprime aussi les exceptions enregistrées dans le document
    If supprimerExceptions Then
        On Error Resume Next
        ActiveDocument.Variables(NOM_VARIABLE_EXCEPTIONS).Delete
        On Error GoTo 0
    End If

    ' [v41] Fin du bloc Undo unique
    On Error Resume Next
    Application.UndoRecord.EndCustomRecord
    ActiveDocument.TrackRevisions = ancienSuiviModifSup
    On Error GoTo 0

End Sub

Sub AutomatiserSiglesEtInfobulles()

                   "puis relancez cette macro.", vbInformation, "Sélection requise"
            GoTo Cleanup
        End If

' =========================================================
' [v80] MACRO AUTONOME : ne fait QUE la détection et le
' surlignage rouge des sigles inconnus, SANS créer aucun lien.
' Utile après un traitement complet + corrections manuelles,
' pour rafraîchir uniquement le rouge sans retoucher aux liens
' déjà en place. Réutilise le dictionnaire (tableau + définitions
' manuelles) et les exceptions, comme le traitement complet, mais
' n'exécute que la partie classification/rouge (pas de collecte
' ni création de liens).
' =========================================================
Sub SurlignerRougeSeul()

    On Error GoTo Cleanup

    ' [v81] FACTORISATION : options via helper commun
    Dim etatOpt As EtatOptionsWord
    etatOpt = DesactiverOptionsTraitement()

    Application.StatusBar = "Lecture des tableaux de sigles..."
    Dim tblDebuts() As Long
    Dim tblFins() As Long
    Dim tblColSigle() As Long
    Dim nbTablesSigle As Long
    Dim dico As Object
    Set dico = ConstruireDictionnaireComplet(tblDebuts, tblFins, tblColSigle, nbTablesSigle)

    If dico.Count = 0 Then
        MsgBox "Aucun sigle détecté (tableau de référence introuvable).", vbInformation
        GoTo Cleanup
    End If

    Dim dicoExceptR As Object
    Set dicoExceptR = ChargerExceptions()

    Dim wStart() As Long
    Dim wFin() As Long
    Dim wTexte() As String
    Dim totalWords As Long
    MaterialiserMots wStart, wFin, wTexte, totalWords

    ' --- Classification : rouge uniquement, pas de lien ---
    Dim compteurRougeR As Long
    compteurRougeR = 0
    Dim idxR As Long
    Dim motCourantR As String
    Dim finPropreR As Long
    Dim wrdR As Range

    For idxR = 1 To totalWords

        If idxR Mod 2000 = 0 Then
            Application.StatusBar = "Analyse : " & idxR & "/" & totalWords & " (" & compteurRougeR & " en rouge)"
            DoEvents
        End If

        motCourantR = RetirerPonctuationFinale(Trim(wTexte(idxR)))

        If EstUnSigle(motCourantR) And Not dicoExceptR.Exists(motCourantR) And Not dico.Exists(motCourantR) Then

            finPropreR = wStart(idxR) + Len(motCourantR)
            Set wrdR = ActiveDocument.Range(wStart(idxR), finPropreR)

            Dim dejaLieR As Boolean
            dejaLieR = False
            On Error Resume Next
            dejaLieR = (wrdR.Hyperlinks.Count > 0)
            On Error GoTo 0

            If Not dejaLieR And Not EstDansTitre(wrdR) Then

                ' [v81] FACTORISATION : exclusion colonne via helper commun
                If Not EstDansColonneSigleExclue(wStart(idxR), wrdR, tblDebuts, tblFins, tblColSigle, nbTablesSigle) Then
                    wrdR.HighlightColorIndex = wdRed
                    compteurRougeR = compteurRougeR + 1
                End If

            End If

        End If

    Next idxR

Cleanup:
    Application.StatusBar = False
    RestaurerOptionsTraitement etatOpt

    If Err.Number <> 0 Then
        MsgBox "Erreur " & Err.Number & " : " & Err.Description, vbCritical
    Else
        MsgBox compteurRougeR & " occurrence(s) surlignée(s) en rouge (sigles sans définition)." & vbCrLf & _
               "Aucun lien n'a été créé ou modifié.", vbInformation, "Terminé"
    End If

End Sub

' =========================================================
' [v89] Détecte les sigles PRÉSENTS dans le(s) tableau(x) de
' référence mais qui n'apparaissent JAMAIS dans le corps du
' document, affiche la liste, et propose de SUPPRIMER ces
' lignes du tableau (ménage avant finalisation).
' =========================================================
Sub SupprimerSiglesInutilises()

    Dim etatOpt As EtatOptionsWord
    etatOpt = DesactiverOptionsTraitement()

    Application.StatusBar = "Lecture des tableaux de sigles..."
    Dim tblDebuts() As Long
    Dim tblFins() As Long
    Dim tblColSigle() As Long
    Dim nbTablesSigle As Long
    Dim dico As Object
    Set dico = ConstruireDictionnaireComplet(tblDebuts, tblFins, tblColSigle, nbTablesSigle)

    If dico.Count = 0 Then
        RestaurerOptionsTraitement etatOpt
        MsgBox "Aucun tableau de sigles trouvé.", vbInformation
        Exit Sub
    End If

    Dim inutilises As Object
    Set inutilises = DetecterSiglesInutilises(dico, tblDebuts, tblFins, tblColSigle, nbTablesSigle)

    RestaurerOptionsTraitement etatOpt

    If inutilises.Count = 0 Then
        MsgBox "Tous les sigles du tableau sont utilisés dans le document. Rien à supprimer.", vbInformation
        Exit Sub
    End If

    Dim listeInutilises As String
    Dim cleI As Variant
    listeInutilises = ""
    For Each cleI In inutilises.Keys
        listeInutilises = listeInutilises & "- " & cleI & " (" & inutilises(cleI) & ")" & vbCrLf
    Next cleI

    Dim repSuppr As VbMsgBoxResult
    repSuppr = MsgBox(inutilises.Count & " sigle(s) présent(s) dans le tableau mais jamais utilisé(s) dans le document :" & vbCrLf & vbCrLf & _
           listeInutilises & vbCrLf & _
           "Supprimer ces lignes du tableau de référence ?", vbYesNo + vbQuestion, "Sigles inutilisés détectés")

    If repSuppr = vbNo Then Exit Sub

    ' --- Suppression des lignes correspondantes dans le(s)
    ' tableau(x), en repartant à l'envers pour ne pas décaler
    ' les index des lignes restantes. ---
    Application.ScreenUpdating = False
    Dim ancienSuiviModifSuppr As Boolean
    On Error Resume Next
    ancienSuiviModifSuppr = ActiveDocument.TrackRevisions
    ActiveDocument.TrackRevisions = False
    Application.UndoRecord.StartCustomRecord "Suppression des sigles inutilisés"
    On Error GoTo 0

    Dim nbSupprimes As Long
    nbSupprimes = 0
    Dim tIdxS As Long

    ' [v91] Nettoyage SYSTÉMATIQUE de la variable de définitions
    ' manuelles pour TOUS les sigles inutilisés détectés, AVANT
    ' même de chercher une ligne de tableau correspondante — couvre
    ' aussi le cas où la ligne a déjà été supprimée MANUELLEMENT
    ' par l'utilisateur dans Word (auquel cas la boucle de
    ' suppression de ligne ci-dessous ne trouverait jamais rien à
    ' faire, et le sigle serait resté indéfiniment dans la
    ' variable persistante sans que ce nettoyage ne soit jamais
    ' déclenché).
    Dim cleNettoyage As Variant
    For Each cleNettoyage In inutilises.Keys
        SupprimerDefinitionManuelle CStr(cleNettoyage)
    Next cleNettoyage

    For tIdxS = 0 To nbTablesSigle - 1

        Dim tblS As Table
        Set tblS = Nothing
        Dim tblCandidat As Table
        On Error Resume Next
        For Each tblCandidat In ActiveDocument.Tables
            If tblCandidat.Range.Start = tblDebuts(tIdxS) Then
                Set tblS = tblCandidat
                Exit For
            End If
        Next tblCandidat
        On Error GoTo 0

        If Not tblS Is Nothing Then

            Dim rIdxS As Long
            For rIdxS = tblS.Rows.Count To 2 Step -1

                Dim texteSigleLigne As String
                texteSigleLigne = ""
                On Error Resume Next
                texteSigleLigne = NettoyerTexteCellule(tblS.Cell(rIdxS, tblColSigle(tIdxS)).Range.Text)
                On Error GoTo 0

                If inutilises.Exists(texteSigleLigne) Then
                    On Error Resume Next
                    tblS.Rows(rIdxS).Delete
                    On Error GoTo 0
                    nbSupprimes = nbSupprimes + 1
                End If

            Next rIdxS

        End If

    Next tIdxS

    On Error Resume Next
    Application.UndoRecord.EndCustomRecord
    ActiveDocument.TrackRevisions = ancienSuiviModifSuppr
    On Error GoTo 0
    Application.ScreenUpdating = True

    MsgBox nbSupprimes & " ligne(s) supprimée(s) du tableau de référence." & vbCrLf & _
           "Pensez à enregistrer le document (Ctrl+S).", vbInformation, "Terminé"

End Sub

' Lien interne (sommaire, renvois) : code du champ contenant \l .
' Les liens générés par la macro ont Address " " (jamais \l).
Private Function EstLienInterne(fld As Field) As Boolean
    Dim code As String
    On Error Resume Next
    code = fld.Code.Text
    On Error GoTo 0
    EstLienInterne = (InStr(code, "\l") > 0)
End Function

' Nombre de liens générés par la macro (hors sommaire / renvois)
Private Function CompterLiensGeneres() As Long
    Dim h As Hyperlink
    Dim n As Long
    On Error Resume Next
    For Each h In ActiveDocument.Hyperlinks
        If Len(h.SubAddress) = 0 Then n = n + 1
    Next h
    On Error GoTo 0
    CompterLiensGeneres = n
End Function

' Niveau de titre pour une numérotation TAPÉE (hors liste Word) : 0 = non
Private Function NiveauTitrePotentiel(txt As String, estGras As Boolean) As Long
    Dim re As Object
    Dim m As Object
    NiveauTitrePotentiel = 0
    If Not FormeTitreOk(txt) Then Exit Function
    Set re = CreateObject("VBScript.RegExp")
    re.IgnoreCase = True
    re.Pattern = "^(\d+(\.\d+){0,3})\s*[\.\)\-–—:]?\s*[A-Za-zÀ-ÿ]"
    If re.Test(txt) Then
        Set m = re.Execute(txt)(0)
        NiveauTitrePotentiel = UBound(Split(m.SubMatches(0), ".")) + 1
        If NiveauTitrePotentiel > 3 Then NiveauTitrePotentiel = 3
        Exit Function
    End If
    re.Pattern = "^[IVX]+\s*[\.\)\-–—]\s*[A-Za-zÀ-ÿ]"
    If re.Test(txt) Then NiveauTitrePotentiel = 1
End Function

' Forme d'un titre : court, pas de ponctuation finale, pas de ": " au milieu
' (écarte "Objet : ...", "Réf : ...", phrases, actions).
Private Function FormeTitreOk(txt As String) As Boolean
    Dim c As String
    FormeTitreOk = False
    If Len(txt) < 3 Or Len(txt) > 100 Then Exit Function
    c = Right(txt, 1)
    If c = "." Or c = ":" Or c = ";" Or c = "," Or c = "?" Or c = "!" Then Exit Function
    If InStr(txt, ":") > 0 Then Exit Function
    If UBound(Split(txt, " ")) + 1 > 10 Then Exit Function
    FormeTitreOk = True
End Function

' Niveau de liste d'un paragraphe numéroté (0 = pas une liste numérotée)
Private Function NiveauListeNumerotee(p As Paragraph) As Long
    Dim lt As Long
    On Error Resume Next
    lt = p.Range.ListFormat.ListType
    If lt = wdListSimpleNumbering Or lt = wdListOutlineNumbering Or lt = wdListMixedNumbering Then
        NiveauListeNumerotee = p.Range.ListFormat.ListLevelNumber
    End If
    On Error GoTo 0
End Function

' [v149] TITRES PAR MODÈLE DU DOCUMENT
' Modèle = un paragraphe déjà en Titre 1 (sinon le paragraphe sélectionné).
' Sont titres : les paragraphes courts de MÊME mise en forme (police, taille,
' gras, italique, couleur) + ceux déjà en Titre 1-3. Le niveau vient du
' décalage à droite : même retrait = Titre 1, 1er palier = Titre 2, 2e = Titre 3.
' Numérotation automatique Word liée aux styles Titre 1-3 (1 / 1.1 / 1.1.1) :
' elle se recalcule seule ; relancer la macro refait tout selon les modèles.

' [v151] Niveau de titre EXISTANT d'un paragraphe : d'abord par son STYLE
' (Titre 1/2/3, Heading 1/2/3, y compris si le niveau hiérarchique a été
' retiré du style), sinon par le niveau hiérarchique 1-3. 0 = pas un titre.
Private Function NiveauTitreExistant(p As Paragraph) As Long
    Dim nm As String
    Dim i As Long
    Dim ol As Long
    On Error Resume Next
    If Len(nomsTitres(1)) = 0 Or reTitreStyle Is Nothing Then
        For i = 1 To 3
            nomsTitres(i) = ActiveDocument.Styles(wdStyleHeading1 - (i - 1)).NameLocal
        Next i
        Set reTitreStyle = CreateObject("VBScript.RegExp")
        reTitreStyle.IgnoreCase = True
        reTitreStyle.Pattern = "^(titre|heading)\s*([1-9])$"
    End If
    nm = p.Style
    For i = 1 To 3
        If StrComp(nm, nomsTitres(i), vbTextCompare) = 0 Then
            NiveauTitreExistant = i
            Exit Function
        End If
    Next i
    If reTitreStyle.Test(nm) Then
        i = CLng(reTitreStyle.Execute(nm)(0).SubMatches(1))
        If i > 3 Then i = 3
        NiveauTitreExistant = i
        Exit Function
    End If
    ol = p.OutlineLevel
    If ol >= 1 And ol <= 3 Then NiveauTitreExistant = ol
    On Error GoTo 0
End Function

Private Function SigMiseEnForme(p As Paragraph) As String
    Dim r As Range
    Set r = p.Range
    On Error Resume Next
    SigMiseEnForme = r.Font.Name & "|" & r.Font.Size & "|" & r.Font.Bold & "|" & r.Font.Italic & "|" & r.Font.Color
    On Error GoTo 0
End Function

' Liste à plusieurs niveaux "SiglesTitres" liée aux styles Titre 1 à 3
Private Function PreparerNumerotationTitres() As ListTemplate
    Dim lt As ListTemplate
    Dim x As ListTemplate
    Dim i As Long
    Dim fmt As String
    On Error Resume Next
    For Each x In ActiveDocument.ListTemplates
        If x.Name = "SiglesTitres" Then Set lt = x: Exit For
    Next x
    On Error GoTo 0
    If lt Is Nothing Then Set lt = ActiveDocument.ListTemplates.Add(OutlineNumbered:=True, Name:="SiglesTitres")
    For i = 1 To 3
        If i = 1 Then fmt = "%1"
        If i = 2 Then fmt = "%1.%2"
        If i = 3 Then fmt = "%1.%2.%3"
        With lt.ListLevels(i)
            .NumberFormat = fmt
            .TrailingCharacter = wdTrailingTab
            .NumberStyle = wdListNumberStyleArabic
            .NumberPosition = 0
            .Alignment = wdListLevelAlignLeft
            .TextPosition = 28.35 * (1 + 0.4 * (i - 1))
            .TabPosition = wdUndefined
            .StartAt = 1
            If i > 1 Then .ResetOnHigher = i - 1
            .LinkedStyle = ActiveDocument.Styles(wdStyleHeading1 - (i - 1)).NameLocal
        End With
    Next i
    Set PreparerNumerotationTitres = lt
End Function

Sub TitresAutomatiques()
    Dim p As Paragraph, pRef As Paragraph
    Dim sigRef As String
    Dim nP As Long, k As Long, n As Long
    Dim txt As String
    Dim niv As Long
    Dim refGauche As Single
    Dim ps() As Long, pOff() As Long, pLvl() As Long, pTxt() As String, pExist() As Boolean, pPref() As Long
    Dim re As Object, mm As Object
    Dim b As Long
    Dim buckets() As Long, nb As Long, j As Long, trouve As Boolean, tmp As Long
    Dim rep As VbMsgBoxResult
    Dim apercu As String
    Dim c1 As Long, c2 As Long, c3 As Long, prev As Long, lv As Long
    Dim num As String
    Dim cAnal As Long, cTab As Long, cStyle As Long, cSig As Long, cForme As Long, cChamp As Long
    Dim dbg As String
    Dim dansTabP As Boolean
    Dim nivP As Long
    nomsTitres(1) = ""

    ' ---- Modèle ----
    For Each p In ActiveDocument.Paragraphs
        If NiveauTitreExistant(p) = 1 Then
            If Len(Trim(Replace(p.Range.Text, vbCr, ""))) > 0 Then
                Set pRef = p
                Exit For
            End If
        End If
    Next p
    If pRef Is Nothing Then
        If Len(Trim(Replace(Selection.Paragraphs(1).Range.Text, vbCr, ""))) > 0 Then
            If MsgBox("Aucun Titre 1 dans le document." & vbCrLf & vbCrLf & "Utiliser le paragraphe sélectionné comme modèle de Titre 1 ?" & vbCrLf & "« " & Left(Trim(Replace(Selection.Paragraphs(1).Range.Text, vbCr, "")), 60) & " »", vbYesNo + vbQuestion, "Titres automatiques") = vbYes Then
                Set pRef = Selection.Paragraphs(1)
            End If
        End If
        If pRef Is Nothing Then
            MsgBox "Mettez un premier titre en style Titre 1 (ou sélectionnez-le), puis relancez.", vbInformation, "Titres automatiques"
            Exit Sub
        End If
    End If
    sigRef = SigMiseEnForme(pRef)
    refGauche = pRef.Format.LeftIndent

    nP = ActiveDocument.Paragraphs.Count
    ReDim ps(1 To nP)
    ReDim pOff(1 To nP)
    ReDim pLvl(1 To nP)
    ReDim pTxt(1 To nP)
    ReDim pExist(1 To nP)
    ReDim pPref(1 To nP)
    Set re = CreateObject("VBScript.RegExp")
    re.Pattern = "^\s*(\d+(\.\d+)+|\d+\s*[\.\)\-–—:]|[IVX]+\s*[\.\)\-–—])\s*"

    ' ---- Repérage ----
    Application.ScreenUpdating = False
    k = 0
    For Each p In ActiveDocument.Paragraphs
        k = k + 1
        If k Mod 300 = 0 Then
            Application.StatusBar = "Recherche des titres : " & k & "/" & nP
            DoEvents
        End If
        On Error Resume Next
        niv = 0
        txt = Trim(Replace(Replace(Replace(p.Range.Text, vbCr, ""), Chr(11), " "), Chr(12), ""))
        If Len(txt) > 0 Then
            cAnal = cAnal + 1
            dansTabP = p.Range.Information(wdWithInTable)
            If dansTabP Then
                cTab = cTab + 1
            Else
                nivP = NiveauTitreExistant(p)
                If nivP > 0 Then
                    ' déjà un titre
                    cStyle = cStyle + 1
                    n = n + 1
                    ps(n) = p.Range.Start
                    pLvl(n) = nivP
                    pOff(n) = Int((p.Format.LeftIndent - refGauche) / 6 + 0.5)
                    pTxt(n) = txt
                    pExist(n) = True
                    pPref(n) = 0
                ElseIf Not FormeTitreOkApresNumero(txt, re) Then
                    ' texte trop long / ponctuation : écarté sans autre appel Word
                    cForme = cForme + 1
                ElseIf p.OutlineLevel = wdOutlineLevelBodyText Then
                    If p.Range.Fields.Count > 0 Then
                        cChamp = cChamp + 1
                    ElseIf SigMiseEnForme(p) = sigRef Then
                        cSig = cSig + 1
                        n = n + 1
                        ps(n) = p.Range.Start
                        pOff(n) = Int((p.Format.LeftIndent - refGauche) / 6 + 0.5)
                        If pOff(n) < 0 Then pOff(n) = 0
                        pTxt(n) = txt
                        pExist(n) = False
                        pPref(n) = 0
                        If re.Test(txt) Then pPref(n) = re.Execute(txt)(0).Length
                    End If
                End If
            End If
        End If
        On Error GoTo 0
    Next p
    Application.StatusBar = False
    Application.ScreenUpdating = True

    dbg = vbCrLf & "--- Détail ---" & vbCrLf & "Modèle : [" & Left(Trim(Replace(pRef.Range.Text, vbCr, "")), 40) & "] style=" & pRef.Style & " mise en forme=" & sigRef & vbCrLf & _
          "Paragraphes non vides : " & cAnal & " (dont en tableau : " & cTab & ")" & vbCrLf & _
          "Déjà niveau Titre 1-3 : " & cStyle & vbCrLf & _
          "Texte normal avec champ/lien (ignorés) : " & cChamp & vbCrLf & _
          "Forme refusée (trop long, ponctuation, « : ») : " & cForme & vbCrLf & "Forme OK ET même mise en forme que le modèle : " & cSig
    If n = 0 Then
        MsgBox "Aucun titre trouvé." & dbg, vbInformation, "Titres automatiques"
        Exit Sub
    End If

    ' ---- Niveaux par paliers de retrait (titres non encore stylés) ----
    ReDim buckets(1 To n + 1)
    nb = 1
    buckets(1) = 0
    For k = 1 To n
        If Not pExist(k) Then
            trouve = False
            For j = 1 To nb
                If buckets(j) = pOff(k) Then trouve = True: Exit For
            Next j
            If Not trouve Then nb = nb + 1: buckets(nb) = pOff(k)
        End If
    Next k
    For k = 1 To nb - 1
        For j = k + 1 To nb
            If buckets(j) < buckets(k) Then tmp = buckets(k): buckets(k) = buckets(j): buckets(j) = tmp
        Next j
    Next k
    For k = 1 To n
        If Not pExist(k) Then
            For j = 1 To nb
                If buckets(j) = pOff(k) Then pLvl(k) = IIf(j > 3, 3, j): Exit For
            Next j
        End If
    Next k

    ' ---- Cohérence (pas de saut de niveau) + numéros prévus ----
    prev = 0
    apercu = ""
    For k = 1 To n
        lv = pLvl(k)
        If lv > prev + 1 Then lv = prev + 1
        pLvl(k) = lv
        prev = lv
        If lv = 1 Then c1 = c1 + 1: c2 = 0: c3 = 0: num = CStr(c1)
        If lv = 2 Then c2 = c2 + 1: c3 = 0: num = c1 & "." & c2
        If lv = 3 Then c3 = c3 + 1: num = c1 & "." & c2 & "." & c3
        If k <= 25 Then apercu = apercu & num & "   " & Left(Mid(pTxt(k), pPref(k) + 1), 55) & vbCrLf
    Next k
    If n > 25 Then apercu = apercu & "... (+" & (n - 25) & ")" & vbCrLf

    rep = MsgBox(n & " titre(s) selon le modèle du document :" & vbCrLf & vbCrLf & apercu & vbCrLf & _
                 "Appliquer Titre 1/2/3 avec numérotation automatique ?" & vbCrLf & _
                 "(les numéros tapés à la main seront retirés)" & dbg, vbYesNo + vbQuestion, "Titres automatiques")
    If rep = vbNo Then Exit Sub

    ' ---- Application (de la fin vers le début) ----
    Application.ScreenUpdating = False
    On Error Resume Next
    Application.UndoRecord.StartCustomRecord "Titres automatiques"
    On Error GoTo 0
    Dim lt As ListTemplate
    Set lt = PreparerNumerotationTitres()
    Dim pr As Paragraph
    For k = n To 1 Step -1
        On Error Resume Next
        Set pr = ActiveDocument.Range(ps(k), ps(k)).Paragraphs(1)
        If pPref(k) > 0 Then ActiveDocument.Range(ps(k), ps(k) + pPref(k)).Delete
        pr.Range.ListFormat.RemoveNumbers
        pr.Range.ParagraphFormat.Reset
        pr.Style = ActiveDocument.Styles(wdStyleHeading1 - (pLvl(k) - 1))
        On Error GoTo 0
    Next k
    On Error Resume Next
    Application.UndoRecord.EndCustomRecord
    On Error GoTo 0
    Application.ScreenUpdating = True

    MsgBox n & " titre(s) appliqué(s), numérotation automatique 1 / 1.1 / 1.1.1." & vbCrLf & _
           "Après modification, relancez ""Titres auto"" puis ""Sommaire"".", vbInformation, "Titres automatiques"
End Sub

' Forme "titre" du texte privé de son numéro tapé
Private Function FormeTitreOkApresNumero(txt As String, re As Object) As Boolean
    Dim s As String
    s = txt
    If re.Test(txt) Then s = Mid(txt, re.Execute(txt)(0).Length + 1)
    FormeTitreOkApresNumero = FormeTitreOk(s)
End Function

' Insère un sommaire cliquable à la position du curseur, ou met à jour
' le sommaire existant.
Sub InsererOuMettreAJourSommaire()
    Dim n As Long, k As Long
    Dim aTitre As Boolean
    Dim p As Paragraph

    For Each p In ActiveDocument.Paragraphs
        If p.OutlineLevel <> wdOutlineLevelBodyText Then aTitre = True: Exit For
    Next p
    If Not aTitre Then
        MsgBox "Aucun titre (styles Titre 1/2/3) dans le document." & vbCrLf & "Lancez d'abord ""Titres auto"".", vbExclamation, "Sommaire"
        Exit Sub
    End If

    On Error Resume Next
    n = ActiveDocument.TablesOfContents.Count
    On Error GoTo 0
    If n > 0 Then
        For k = 1 To n
            ActiveDocument.TablesOfContents(k).Update
        Next k
        MsgBox "Sommaire mis à jour.", vbInformation, "Sommaire"
    Else
        On Error Resume Next
        ActiveDocument.TablesOfContents.Add Range:=Selection.Range, _
            UseHeadingStyles:=True, UpperHeadingLevel:=1, LowerHeadingLevel:=3, _
            UseHyperlinks:=True, HidePageNumbersInWeb:=True
        If Err.Number <> 0 Then
            MsgBox "Création du sommaire impossible : " & Err.Description, vbExclamation, "Sommaire"
        Else
            MsgBox "Sommaire inséré à la position du curseur (liens cliquables : Ctrl+clic).", vbInformation, "Sommaire"
        End If
        On Error GoTo 0
    End If
End Sub

' [v152] UNIFORMISER LES POLICES : police unique partout (titres inclus,
' leur taille est conservée) ; taille unique pour le texte hors titres.
' Sommaire (TOC) toujours exclu ; tableaux au choix.
Sub UniformiserPolices()
    Dim police As String, tailleTxt As String, taille As Single
    Dim inclTab As Boolean
    Dim p As Paragraph
    Dim k As Long, nTxt As Long, nTit As Long, nSkip As Long, nP As Long
    Dim tocS() As Long, tocE() As Long, nToc As Long, t As Long
    Dim dansToc As Boolean, estTit As Boolean, nm As String
    Dim ancienSuivi As Boolean

    police = InputBox("Police à appliquer (texte ET titres) :", "Uniformiser les polices", ActiveDocument.Styles(wdStyleNormal).Font.Name)
    If Len(Trim(police)) = 0 Then Exit Sub
    tailleTxt = InputBox("Taille du texte hors titres (pt) :" & vbCrLf & "(la taille des titres n'est pas modifiée)", "Uniformiser les polices", CStr(ActiveDocument.Styles(wdStyleNormal).Font.Size))
    If Len(Trim(tailleTxt)) = 0 Then Exit Sub
    taille = Val(Replace(tailleTxt, ",", "."))
    If taille < 1 Or taille > 200 Then
        MsgBox "Taille invalide.", vbExclamation
        Exit Sub
    End If
    inclTab = (MsgBox("Inclure aussi le contenu des TABLEAUX ?" & vbCrLf & "(Non = tableaux inchangés, dont le tableau des sigles)", vbYesNo + vbQuestion, "Uniformiser les polices") = vbYes)

    ' Plages des sommaires
    ReDim tocS(0 To 20)
    ReDim tocE(0 To 20)
    On Error Resume Next
    For t = 1 To ActiveDocument.TablesOfContents.Count
        If nToc > 20 Then Exit For
        tocS(nToc) = ActiveDocument.TablesOfContents(t).Range.Start
        tocE(nToc) = ActiveDocument.TablesOfContents(t).Range.End
        nToc = nToc + 1
    Next t
    On Error GoTo 0

    ancienSuivi = False
    On Error Resume Next
    ancienSuivi = ActiveDocument.TrackRevisions
    ActiveDocument.TrackRevisions = False
    Application.UndoRecord.StartCustomRecord "Uniformiser les polices"
    On Error GoTo 0
    Application.ScreenUpdating = False

    nP = ActiveDocument.Paragraphs.Count
    For Each p In ActiveDocument.Paragraphs
        k = k + 1
        If k Mod 200 = 0 Then
            Application.StatusBar = "Uniformisation : " & k & "/" & nP
            DoEvents
        End If

        On Error Resume Next
        dansToc = False
        For t = 0 To nToc - 1
            If p.Range.Start >= tocS(t) And p.Range.Start < tocE(t) Then dansToc = True: Exit For
        Next t
        nm = p.Style
        If InStr(1, nm, "TOC", vbTextCompare) > 0 Or InStr(1, nm, "TM ", vbTextCompare) > 0 Or InStr(1, nm, "matières", vbTextCompare) > 0 Then dansToc = True

        If dansToc Then
            nSkip = nSkip + 1
        ElseIf (Not inclTab) And p.Range.Information(wdWithInTable) = True Then
            nSkip = nSkip + 1
        Else
            estTit = (NiveauTitreExistant(p) > 0)
            If estTit Then
                p.Range.Font.Name = police
                nTit = nTit + 1
            Else
                p.Range.Font.Name = police
                p.Range.Font.Size = taille
                nTxt = nTxt + 1
            End If
        End If
        On Error GoTo 0
    Next p

    Application.StatusBar = False
    Application.ScreenUpdating = True
    On Error Resume Next
    Application.UndoRecord.EndCustomRecord
    ActiveDocument.TrackRevisions = ancienSuivi
    On Error GoTo 0

    MsgBox nTxt & " paragraphe(s) : " & police & " " & taille & " pt." & vbCrLf & _
           nTit & " titre(s) : " & police & " (taille conservée)." & vbCrLf & _
           nSkip & " ignoré(s) (sommaire" & IIf(inclTab, "", ", tableaux") & ")." & vbCrLf & vbCrLf & _
           "Ctrl+Z annule tout.", vbInformation, "Polices uniformisées"
End Sub

Sub Ruban_UniformiserPolices(control As IRibbonControl)
    UniformiserPolices
End Sub

' [v153] Vrai si le paragraphe contient une légende "Figure" (champ SEQ)
Private Function ParaALegendeFigure(p As Paragraph) As Boolean
    Dim f As Field
    On Error Resume Next
    For Each f In p.Range.Fields
        If f.Type = wdFieldSequence Then
            If InStr(1, f.Code.Text, "Figure", vbTextCompare) > 0 Then ParaALegendeFigure = True: Exit For
        End If
    Next f
    On Error GoTo 0
End Function

' [v153] LÉGENDES AUTOMATIQUES DES IMAGES + TABLE DES ILLUSTRATIONS
' Légende dessous, libellé "Figure", texte à compléter, images flottantes
' converties en ligne. Relancer = n'ajoute que les légendes manquantes.
Sub LegendesImages()
    Dim i As Long
    Dim shp As Shape
    Dim ils As InlineShape
    Dim nConv As Long, nAjout As Long, nDeja As Long, nPetit As Long, nErr As Long
    Dim pos() As Long, nPos As Long
    Dim rngImg As Range
    Dim pr As Paragraph
    Dim pSuiv As Paragraph, pPrec As Paragraph
    Dim ancienSuivi As Boolean
    Dim f As Field
    Dim tdi As TableOfFigures
    Dim posCurseur As Long
    Dim aTable As Boolean
    Dim repTable As VbMsgBoxResult
    Dim bmNom As String

    bmNom = "_CurseurFigures"
    On Error Resume Next
    ActiveDocument.Bookmarks.Add bmNom, Selection.Range
    aTable = (ActiveDocument.TablesOfFigures.Count > 0)
    On Error GoTo 0

    ancienSuivi = False
    On Error Resume Next
    ancienSuivi = ActiveDocument.TrackRevisions
    ActiveDocument.TrackRevisions = False
    Application.UndoRecord.StartCustomRecord "Légendes des images"
    On Error GoTo 0
    Application.ScreenUpdating = False

    ' --- Libellé "Figure" disponible ---
    Dim lblOk As Boolean
    On Error Resume Next
    lblOk = (Len(CaptionLabels("Figure").Name) > 0)
    If Not lblOk Then CaptionLabels.Add "Figure"
    On Error GoTo 0

    ' --- 1. Images flottantes -> en ligne ---
    For i = ActiveDocument.Shapes.Count To 1 Step -1
        Set shp = Nothing
        On Error Resume Next
        Set shp = ActiveDocument.Shapes(i)
        If Not shp Is Nothing Then
            If shp.Anchor.StoryType = wdMainTextStory Then
                If shp.Type = msoPicture Or shp.Type = msoLinkedPicture Then
                    If shp.Width >= 28 And shp.Height >= 28 Then
                        Err.Clear
                        shp.ConvertToInlineShape
                        If Err.Number = 0 Then nConv = nConv + 1 Else nErr = nErr + 1
                    End If
                End If
            End If
        End If
        On Error GoTo 0
    Next i

    ' --- 2. Positions des images en ligne à légender ---
    ReDim pos(0 To ActiveDocument.InlineShapes.Count + 1)
    For Each ils In ActiveDocument.InlineShapes
        On Error Resume Next
        If ils.Type = wdInlineShapePicture Or ils.Type = wdInlineShapeLinkedPicture Then
            If ils.Width < 28 Or ils.Height < 28 Then
                nPetit = nPetit + 1
            Else
                Set pr = ils.Range.Paragraphs(1)
                Set pSuiv = Nothing
                Set pPrec = Nothing
                Set pSuiv = pr.Next
                Set pPrec = pr.Previous
                If ParaALegendeFigure(pr) Then
                    nDeja = nDeja + 1
                ElseIf Not pSuiv Is Nothing And ParaALegendeFigure(pSuiv) Then
                    nDeja = nDeja + 1
                ElseIf Not pPrec Is Nothing And ParaALegendeFigure(pPrec) Then
                    nDeja = nDeja + 1
                Else
                    pos(nPos) = ils.Range.Start
                    nPos = nPos + 1
                End If
            End If
        End If
        On Error GoTo 0
    Next ils

    ' --- 3. Légendes, de la fin vers le début ---
    For i = nPos - 1 To 0 Step -1
        Application.StatusBar = "Légendes : " & (nPos - i) & "/" & nPos
        Set rngImg = ActiveDocument.Range(pos(i), pos(i) + 1)
        On Error Resume Next
        Err.Clear
        rngImg.InsertCaption Label:="Figure", Title:=" : ", Position:=wdCaptionPositionBelow, ExcludeLabel:=False
        If Err.Number <> 0 Then
            nErr = nErr + 1
            Err.Clear
        Else
            Set pr = rngImg.Paragraphs(1)
            pr.KeepWithNext = True
            Set pSuiv = pr.Next
            If Not pSuiv Is Nothing Then pSuiv.Alignment = wdAlignParagraphCenter
            nAjout = nAjout + 1
        End If
        On Error GoTo 0
    Next i
    Application.StatusBar = False

    ' --- 4. Renumérote les légendes ---
    On Error Resume Next
    For Each f In ActiveDocument.Fields
        If f.Type = wdFieldSequence Then f.Update
    Next f
    On Error GoTo 0

    ' --- 5. Table des illustrations ---
    Dim msgTable As String
    If aTable Then
        On Error Resume Next
        For Each tdi In ActiveDocument.TablesOfFigures
            tdi.Update
        Next tdi
        On Error GoTo 0
        msgTable = "Table des illustrations mise à jour."
    ElseIf nAjout + nDeja > 0 Then
        Application.ScreenUpdating = True
        repTable = MsgBox("Insérer la table des illustrations à la position du curseur ?", vbYesNo + vbQuestion, "Table des illustrations")
        Application.ScreenUpdating = False
        If repTable = vbYes Then
            On Error Resume Next
            Err.Clear
            Set rngImg = ActiveDocument.Bookmarks(bmNom).Range
            ActiveDocument.TablesOfFigures.Add Range:=rngImg, Caption:="Figure", IncludeLabel:=True, UseHyperlinks:=True, HidePageNumbersInWeb:=True
            If Err.Number <> 0 Then
                msgTable = "Table non insérée : " & Err.Description
            Else
                msgTable = "Table des illustrations insérée au curseur (Ctrl+clic = lien)."
            End If
            On Error GoTo 0
        End If
    End If

    On Error Resume Next
    ActiveDocument.Bookmarks(bmNom).Delete
    Application.UndoRecord.EndCustomRecord
    ActiveDocument.TrackRevisions = ancienSuivi
    On Error GoTo 0
    Application.ScreenUpdating = True

    MsgBox nAjout & " légende(s) ajoutée(s) (« Figure N : » à compléter)." & vbCrLf & _

' [v147] DIAGNOSTIC TITRES : écrit dans un NOUVEAU document les paragraphes
' courts "suspects" (liste, style titre, gras...) avec leurs propriétés Word,
' pour comprendre pourquoi "Titres auto" ne les détecte pas.
Sub DiagnostiquerTitres()
    Dim p As Paragraph
    Dim docSrc As Document, docD As Document
    Dim k As Long, n As Long, lignes As Long
    Dim txt As String, st As String, ls As String
    Dim lt As Long, lv As Long, ol As Long
    Dim gras As Boolean, dansTab As Boolean
    Dim sortie As String

    Set docSrc = ActiveDocument
    sortie = "N°" & vbTab & "Style" & vbTab & "ListType" & vbTab & "ListString" & vbTab & "NivListe" & vbTab & "Outline" & vbTab & "Gras" & vbTab & "Tableau" & vbTab & "Texte" & vbCrLf
    For Each p In docSrc.Paragraphs
        k = k + 1
        On Error Resume Next
        txt = Trim(Replace(Replace(p.Range.Text, vbCr, ""), Chr(7), ""))
        If Len(txt) > 0 And Len(txt) <= 100 Then
            st = p.Style
            lt = p.Range.ListFormat.ListType
            ls = p.Range.ListFormat.ListString
            lv = p.Range.ListFormat.ListLevelNumber
            ol = p.OutlineLevel
            gras = (p.Range.Font.Bold = True)
            dansTab = p.Range.Information(wdWithInTable)
            If lt <> 0 Or ol <> wdOutlineLevelBodyText Or gras Or InStr(1, st, "titre", vbTextCompare) > 0 Or InStr(1, st, "head", vbTextCompare) > 0 Then
                If Not dansTab Then
                    sortie = sortie & k & vbTab & st & vbTab & lt & vbTab & "[" & ls & "]" & vbTab & lv & vbTab & ol & vbTab & gras & vbTab & dansTab & vbTab & Left(txt, 45) & vbCrLf
                    n = n + 1
                    If n >= 150 Then Exit For
                End If
            End If
        End If
        On Error GoTo 0
    Next p
    Set docD = Documents.Add
    docD.Content.Text = sortie
    docD.Content.Font.Size = 8
    docD.Content.Font.Name = "Consolas"
    MsgBox n & " paragraphe(s) listé(s) dans le nouveau document. Photographiez-le ou copiez-le.", vbInformation, "Diagnostic titres"
End Sub
